# 安全与风控设计文档

## 1. 概述

### 1.1 设计目标
- **资金安全**: 多层风控保护,防止大额亏损
- **数据安全**: 敏感信息加密存储
- **系统安全**: 防止未授权访问
- **交易安全**: 二次确认、风控检查
- **可审计**: 完整的操作日志

### 1.2 风险类型
- **市场风险**: 价格波动导致的亏损
- **操作风险**: 错误操作、系统故障
- **技术风险**: 代码漏洞、数据错误
- **合规风险**: 违反交易规则

### 2.2 风控规则详细设计

#### 2.2.1 仓位控制

```python
class PositionRiskControl:
    """仓位风控"""

    def __init__(self, config: dict):
        self.max_single_position_pct = config.get("max_single_position_pct", 0.20)  # 20%
        self.max_total_position_pct = config.get("max_total_position_pct", 0.80)    # 80%
        self.min_cash_reserve_pct = config.get("min_cash_reserve_pct", 0.20)        # 20%

    def check_single_position_limit(
        self,
        symbol: str,
        order_amount: float,
        current_position_value: float,
        total_value: float
    ) -> tuple[bool, str]:
        """检查单股仓位限制"""

        new_position_value = current_position_value + order_amount
        position_pct = new_position_value / total_value

        if position_pct > self.max_single_position_pct:
            return False, (
                f"单股仓位超限: {symbol} 将占 {position_pct:.1%}, "
                f"超过限制 {self.max_single_position_pct:.1%}"
            )

        return True, ""

    def check_total_position_limit(
        self,
        order_amount: float,
        current_total_position: float,
        total_value: float
    ) -> tuple[bool, str]:
        """检查总仓位限制"""

        new_total_position = current_total_position + order_amount
        position_pct = new_total_position / total_value

        if position_pct > self.max_total_position_pct:
            return False, (
                f"总仓位超限: 将占 {position_pct:.1%}, "
                f"超过限制 {self.max_total_position_pct:.1%}"
            )

        return True, ""

    def check_cash_reserve(
        self,
        order_amount: float,
        cash: float,
        total_value: float
    ) -> tuple[bool, str]:
        """检查现金储备"""

        remaining_cash = cash - order_amount
        cash_pct = remaining_cash / total_value

        if cash_pct < self.min_cash_reserve_pct:
            return False, (
                f"现金储备不足: 剩余 {cash_pct:.1%}, "
                f"低于要求 {self.min_cash_reserve_pct:.1%}"
            )

        return True, ""
```

#### 2.2.2 亏损控制

```python
class LossRiskControl:
    """亏损风控"""

    def __init__(self, config: dict):
        self.max_daily_loss_pct = config.get("max_daily_loss_pct", 0.03)          # 3%
        self.max_single_stock_loss_pct = config.get("max_single_stock_loss_pct", 0.05)  # 5%
        self.max_consecutive_losses = config.get("max_consecutive_losses", 3)

    def check_daily_loss_limit(
        self,
        daily_loss: float,
        total_value: float
    ) -> tuple[bool, str]:
        """检查日内亏损限制"""

        daily_loss_pct = abs(daily_loss) / total_value

        if daily_loss_pct >= self.max_daily_loss_pct:
            return False, (
                f"日内亏损达到限制: {daily_loss_pct:.2%} >= {self.max_daily_loss_pct:.2%}, "
                f"暂停交易"
            )

        return True, ""

    def check_stop_loss_triggered(
        self,
        positions: Dict[str, Position]
    ) -> List[str]:
        """检查止损触发"""

        triggered = []

        for symbol, pos in positions.items():
            # 检查自定义止损
            if pos.stop_loss and pos.current_price <= pos.stop_loss:
                triggered.append(symbol)
                logger.warning(
                    f"止损触发: {symbol} 当前价 {pos.current_price} "
                    f"<= 止损价 {pos.stop_loss}"
                )
                continue

            # 检查默认止损
            loss_pct = (pos.current_price - pos.avg_cost) / pos.avg_cost

            if loss_pct <= -self.max_single_stock_loss_pct:
                triggered.append(symbol)
                logger.warning(
                    f"默认止损触发: {symbol} 亏损 {loss_pct:.2%} "
                    f"<= {-self.max_single_stock_loss_pct:.2%}"
                )

        return triggered

    def check_consecutive_losses(
        self,
        consecutive_losses: int
    ) -> tuple[bool, str]:
        """检查连续亏损"""

        if consecutive_losses >= self.max_consecutive_losses:
            return False, (
                f"连续亏损 {consecutive_losses} 次, "
                f"达到限制 {self.max_consecutive_losses} 次, 暂停交易"
            )

        return True, ""
```

```python
class FrequencyRiskControl:
    """交易频率风控"""

    def __init__(self, config: dict):
        self.max_daily_trades = config.get("max_daily_trades", 20)
        self.min_hold_period = config.get("min_hold_period", 1)  # 天
        self.max_turnover_rate = config.get("max_turnover_rate", 2.0)  # 日换手率

    def check_daily_trade_limit(
        self,
        today_trades_count: int
    ) -> tuple[bool, str]:
        """检查日内交易次数限制"""

        if today_trades_count >= self.max_daily_trades:
            return False, (
                f"日内交易次数达到限制: {today_trades_count} "
                f">= {self.max_daily_trades}"
            )

        return True, ""

    def check_min_hold_period(
        self,
        symbol: str,
        first_buy_at: datetime
    ) -> tuple[bool, str]:
        """检查最小持仓时间"""

        hold_days = (datetime.now() - first_buy_at).days

        if hold_days < self.min_hold_period:
            return False, (
                f"持仓时间不足: {symbol} 持有 {hold_days} 天, "
                f"低于要求 {self.min_hold_period} 天"
            )

        return True, ""

    def check_turnover_rate(
        self,
        today_volume: float,
        total_value: float
    ) -> tuple[bool, str]:
        """检查换手率"""

        turnover_rate = today_volume / total_value

        if turnover_rate > self.max_turnover_rate:
            return False, (
                f"日换手率过高: {turnover_rate:.2f} > {self.max_turnover_rate:.2f}"
            )

        return True, ""
```

#### 2.2.4 市场条件检查

```python
class MarketRiskControl:
    """市场风控"""

    def check_trading_hours(self) -> tuple[bool, str]:
        """检查交易时间"""
        now = datetime.now()
        hour = now.hour
        minute = now.minute
        weekday = now.weekday()

        # 非交易日
        if weekday >= 5:  # 周六、周日
            return False, "非交易日"

        # A股交易时间: 9:30-11:30, 13:00-15:00
        morning_start = (9, 30)
        morning_end = (11, 30)
        afternoon_start = (13, 0)
        afternoon_end = (15, 0)

        in_morning = (hour, minute) >= morning_start and (hour, minute) <= morning_end
        in_afternoon = (hour, minute) >= afternoon_start and (hour, minute) <= afternoon_end

        if not (in_morning or in_afternoon):
            return False, "非交易时间"

        return True, ""

    def check_market_volatility(
        self,
        volatility: float,
        threshold: float = 0.05
    ) -> tuple[bool, str]:
        """检查市场波动率"""

        if volatility > threshold:
            return False, (
                f"市场波动率过高: {volatility:.2%} > {threshold:.2%}, "
                f"建议暂停交易"
            )

        return True, ""

    def check_circuit_breaker(
        self,
        index_change_pct: float
    ) -> tuple[bool, str]:
        """检查熔断"""

        # A股熔断机制 (假设)
        if abs(index_change_pct) >= 0.05:  # 5%
            return False, f"触发熔断: 指数变动 {index_change_pct:.2%}"

        return True, ""
```

### 2.3 风控管理器集成

```python
class RiskManager:
    """风控管理器(完整版)"""

    def __init__(self, config: dict):
        self.config = config

        # 初始化各风控模块
        self.position_control = PositionRiskControl(config.get("position", {}))
        self.loss_control = LossRiskControl(config.get("loss", {}))
        self.frequency_control = FrequencyRiskControl(config.get("frequency", {}))
        self.market_control = MarketRiskControl()

        # 风控开关
        self.enabled = config.get("enabled", True)
        self.emergency_stop = False

    def pre_order_check(
        self,
        order: Order,
        account: Account,
        positions: Dict[str, Position],
        market_data: dict
    ) -> tuple[bool, str]:
        """订单前置风控检查"""

        if not self.enabled:
            return True, "风控已禁用"

        if self.emergency_stop:
            return False, "紧急停止交易"

        # 1. 市场条件检查
        passed, reason = self.market_control.check_trading_hours()
        if not passed:
            return False, reason

        # 2. 亏损检查
        if order.side == OrderSide.BUY:
            passed, reason = self.loss_control.check_daily_loss_limit(
                account.daily_loss,
                account.total_value
            )
            if not passed:
                return False, reason

            passed, reason = self.loss_control.check_consecutive_losses(
                account.consecutive_losses
            )
            if not passed:
                return False, reason

        # 3. 仓位检查
        if order.side == OrderSide.BUY:
            order_amount = order.quantity * (order.price or market_data[order.symbol]["price"])

            # 单股仓位
            current_position_value = (
                positions[order.symbol].market_value
                if order.symbol in positions else 0
            )
            passed, reason = self.position_control.check_single_position_limit(
                order.symbol,
                order_amount,
                current_position_value,
                account.total_value
            )
            if not passed:
                return False, reason

            # 总仓位
            passed, reason = self.position_control.check_total_position_limit(
                order_amount,
                account.position_value,
                account.total_value
            )
            if not passed:
                return False, reason

            # 现金储备
            passed, reason = self.position_control.check_cash_reserve(
                order_amount,
                account.cash,
                account.total_value
            )
            if not passed:
                return False, reason

        # 4. 频率检查
        # ... (根据需要添加)

        return True, "风控检查通过"

    def post_trade_check(
        self,
        positions: Dict[str, Position]
    ) -> List[str]:
        """交易后风控检查(止损触发等)"""

        # 检查止损触发
        triggered = self.loss_control.check_stop_loss_triggered(positions)

        return triggered

    def enable_emergency_stop(self, reason: str):
        """启用紧急停止"""
        self.emergency_stop = True
        logger.critical(f"紧急停止交易: {reason}")

    def disable_emergency_stop(self):
        """解除紧急停止"""
        self.emergency_stop = False
        logger.info("解除紧急停止")
```

```python
from cryptography.fernet import Fernet
import base64
from typing import Union

class SecurityManager:
    """安全管理器"""

    def __init__(self, secret_key: str = None):
        if secret_key is None:
            secret_key = Fernet.generate_key()
        self.cipher = Fernet(secret_key)

    def encrypt(self, data: Union[str, bytes]) -> str:
        """加密数据"""
        if isinstance(data, str):
            data = data.encode()
        encrypted = self.cipher.encrypt(data)
        return base64.b64encode(encrypted).decode()

    def decrypt(self, encrypted_data: str) -> str:
        """解密数据"""
        encrypted = base64.b64decode(encrypted_data)
        decrypted = self.cipher.decrypt(encrypted)
        return decrypted.decode()

# 使用示例
security = SecurityManager()

# 加密 API Key
api_key = "your_secret_api_key"
encrypted_key = security.encrypt(api_key)

# 存储加密后的 key
config["broker"]["api_key"] = encrypted_key

# 使用时解密
api_key = security.decrypt(config["broker"]["api_key"])
```

```python
import re

class SensitiveDataFilter:
    """敏感数据过滤器"""

    @staticmethod
    def mask_api_key(text: str) -> str:
        """脱敏 API Key"""
        pattern = r'(api_key["\']?\s*[:=]\s*["\']?)([a-zA-Z0-9]{20,})'
        return re.sub(pattern, r'\1***MASKED***', text)

    @staticmethod
    def mask_password(text: str) -> str:
        """脱敏密码"""
        pattern = r'(password["\']?\s*[:=]\s*["\']?)([^"\']+)'
        return re.sub(pattern, r'\1***MASKED***', text)

    @staticmethod
    def filter_log(message: str) -> str:
        """过滤日志中的敏感信息"""
        message = SensitiveDataFilter.mask_api_key(message)
        message = SensitiveDataFilter.mask_password(message)
        return message

# 应用到日志系统
from loguru import logger

logger.remove()
logger.add(
    sys.stdout,
    format="{time} {level} {message}",
    filter=lambda record: SensitiveDataFilter.filter_log(record["message"])
)
```

```python
def get_audit_logs(
    start_date: datetime,
    end_date: datetime,
    module: str = None,
    level: str = None
) -> List[Log]:
    """查询审计日志"""
    query = session.query(Log).filter(
        Log.timestamp >= start_date,
        Log.timestamp <= end_date
    )

    if module:
        query = query.filter(Log.module == module)

    if level:
        query = query.filter(Log.level == level)

    return query.order_by(Log.timestamp.desc()).all()
```

```python
class RiskMonitor:
    """风险监控"""

    def __init__(self, alert_manager):
        self.alert_manager = alert_manager
        self.thresholds = {
            "daily_loss_warning": 0.02,      # 日亏损预警 2%
            "position_concentration": 0.30,   # 持仓集中度预警 30%
            "drawdown_warning": 0.10          # 回撤预警 10%
        }

    def monitor_account(self, account: Account):
        """监控账户状态"""

        # 日亏损预警
        daily_loss_pct = abs(account.daily_pnl) / account.total_value
        if daily_loss_pct >= self.thresholds["daily_loss_warning"]:
            self.alert_manager.send_alert(
                level="WARNING",
                title="日亏损预警",
                message=f"日内亏损 {daily_loss_pct:.2%}, 接近限制"
            )

    def monitor_positions(self, positions: Dict[str, Position]):
        """监控持仓"""

        # 持仓集中度检查
        if positions:
            total_value = sum(pos.market_value for pos in positions.values())
            for symbol, pos in positions.items():
                concentration = pos.market_value / total_value

                if concentration >= self.thresholds["position_concentration"]:
                    self.alert_manager.send_alert(
                        level="WARNING",
                        title="持仓集中度预警",
                        message=f"{symbol} 占仓位 {concentration:.1%}"
                    )
```

## 8. 总结

安全与风控设计要点:
1. **多层风控**: 硬编码规则 + 策略级风控 + 实时监控
2. **不可绕过**: 核心风控规则硬编码,保护资金安全
3. **数据安全**: 敏感信息加密、日志脱敏
4. **完整审计**: 所有操作可追溯
5. **实时监控**: 及时发现并处理风险

风控是量化交易系统的生命线,必须谨慎设计、严格执行。

# 回测引擎设计文档

## 1. 模块概述

### 1.1 核心职责
回测引擎负责在历史数据上模拟策略执行,评估策略表现:
- 策略回测执行
- 订单模拟撮合
- 仓位管理模拟
- 交易成本计算
- 绩效指标统计
- 回测报告生成

### 1.2 设计目标
- **真实性**: 尽可能还原真实交易环境
- **准确性**: 精确计算交易成本和滑点
- **全面性**: 提供完整的绩效评估指标
- **可对比**: 支持与基准对比
- **可视化**: 清晰展示回测结果

### 1.3 关键假设
- **价格数据**: 使用复权后的日线数据
- **成交假设**: 当日信号,次日开盘价成交
- **滑点模型**: 固定滑点或基于成交量的动态滑点
- **交易限制**: 遵守涨跌停板限制
- **资金假设**: 不考虑融资融券

### 3.1 回测引擎核心

```python
from typing import Dict, List, Optional
from datetime import datetime, timedelta
import pandas as pd
from enum import Enum

class BacktestMode(Enum):
    """回测模式"""
    BAR = "bar"           # 日线回测
    TICK = "tick"         # tick 级回测（暂不实现）

class BacktestEngine:
    """回测引擎"""

    def __init__(self, config: Dict):
        self.config = config
        self.mode = BacktestMode(config.get("mode", "bar"))

        # 初始化组件
        self.data_engine = config["data_engine"]
        self.strategy = config["strategy"]
        self.portfolio = Portfolio(
            initial_capital=config.get("initial_capital", 1000000),
            commission_model=self._create_commission_model(),
            slippage_model=self._create_slippage_model()
        )

        # 回测状态
        self.current_date = None
        self.is_running = False

        # 结果记录
        self.daily_records = []
        self.trade_records = []

    def run(
        self,
        symbols: List[str],
        start_date: str,
        end_date: str
    ) -> Dict:
        """执行回测"""
        logger.info(f"Starting backtest: {start_date} to {end_date}")
        self.is_running = True

        try:
            # 1. 加载数据
            data = self._load_data(symbols, start_date, end_date)

            # 2. 初始化策略
            self.strategy.initialize(
                context=BacktestContext(
                    data=data,
                    portfolio=self.portfolio,
                    config=self.config
                )
            )

            # 3. 获取所有交易日
            all_dates = sorted(set(data[symbols[0]].index))

            # 4. 时间循环
            for date in all_dates:
                self.current_date = date
                self._run_single_day(data, date)

            # 5. 计算绩效指标
            metrics = self._calculate_metrics()

            # 6. 生成报告
            report = self._generate_report(metrics)

            logger.info("Backtest completed successfully")
            return {
                "metrics": metrics,
                "report": report,
                "daily_records": self.daily_records,
                "trade_records": self.trade_records
            }

        except Exception as e:
            logger.error(f"Backtest failed: {e}")
            raise

        finally:
            self.is_running = False

    def _run_single_day(self, data: Dict[str, pd.DataFrame], date: datetime):
        """执行单日回测"""
        # 1. 更新市场数据（截止到当前日期）
        current_data = {
            symbol: df.loc[:date]
            for symbol, df in data.items()
        }

        # 2. 策略生成信号
        signals = self.strategy.generate_signals(current_data, date)

        # 3. 处理信号生成订单
        for signal in signals:
            self._process_signal(signal, current_data)

        # 4. 订单撮合（使用次日开盘价）
        next_date = self._get_next_trading_day(date, data)
        if next_date:
            self._execute_orders(data, next_date)

        # 5. 更新持仓市值（使用当日收盘价）
        self.portfolio.update_market_value(current_data, date)

        # 6. 记录每日状态
        self._record_daily_status(date)

    def _process_signal(self, signal: Dict, data: Dict[str, pd.DataFrame]):
        """处理交易信号"""
        symbol = signal["symbol"]
        action = signal["action"]  # BUY / SELL
        quantity = signal.get("quantity")
        price = signal.get("price")

        # 创建订单
        if action == "BUY":
            self.portfolio.create_order(
                symbol=symbol,
                quantity=quantity,
                order_type="market",
                direction="long"
            )
        elif action == "SELL":
            self.portfolio.create_order(
                symbol=symbol,
                quantity=quantity,
                order_type="market",
                direction="short"
            )

    def _execute_orders(self, data: Dict[str, pd.DataFrame], date: datetime):
        """执行订单"""
        for order in self.portfolio.pending_orders:
            symbol = order.symbol

            # 获取次日开盘价
            if date in data[symbol].index:
                open_price = data[symbol].loc[date, "open"]

                # 应用滑点
                execution_price = self.portfolio.slippage_model.apply(
                    price=open_price,
                    direction=order.direction,
                    volume=order.quantity
                )

                # 检查涨跌停
                if self._check_limit_price(data[symbol], date, execution_price):
                    # 成交
                    trade = self.portfolio.execute_order(order, execution_price, date)
                    self.trade_records.append(trade)
                else:
                    # 无法成交（涨跌停）
                    order.status = "cancelled"
                    logger.warning(f"Order cancelled due to limit price: {order}")

        # 清理已处理订单
        self.portfolio.clear_executed_orders()

    def _check_limit_price(
        self,
        df: pd.DataFrame,
        date: datetime,
        price: float
    ) -> bool:
        """检查是否触及涨跌停板"""
        prev_close = df.loc[:date].iloc[-2]["close"]

        # A股涨跌停限制：±10%（ST股票 ±5%）
        upper_limit = prev_close * 1.10
        lower_limit = prev_close * 0.90

        return lower_limit <= price <= upper_limit

    def _load_data(
        self,
        symbols: List[str],
        start_date: str,
        end_date: str
    ) -> Dict[str, pd.DataFrame]:
        """加载历史数据"""
        data = {}

        for symbol in symbols:
            df = self.data_engine.get_daily_data(
                symbol=symbol,
                start_date=start_date,
                end_date=end_date,
                adjust="qfq"  # 前复权
            )

            # 计算技术指标
            df = self._prepare_indicators(df)

            data[symbol] = df

        return data

    def _prepare_indicators(self, df: pd.DataFrame) -> pd.DataFrame:
        """预计算技术指标"""
        # 根据策略需求计算指标
        indicators = self.strategy.get_required_indicators()

        for indicator_name, params in indicators.items():
            indicator = IndicatorRegistry.get(indicator_name, **params)
            df = indicator.calculate(df)

        return df

    def _get_next_trading_day(
        self,
        date: datetime,
        data: Dict[str, pd.DataFrame]
    ) -> Optional[datetime]:
        """获取下一个交易日"""
        # 使用第一个股票的索引作为交易日历
        symbol = list(data.keys())[0]
        dates = data[symbol].index

        try:
            current_idx = dates.get_loc(date)
            if current_idx + 1 < len(dates):
                return dates[current_idx + 1]
        except KeyError:
            pass

        return None

    def _record_daily_status(self, date: datetime):
        """记录每日状态"""
        self.daily_records.append({
            "date": date,
            "cash": self.portfolio.cash,
            "position_value": self.portfolio.position_value,
            "total_value": self.portfolio.total_value,
            "returns": self.portfolio.returns,
            "positions": self.portfolio.get_positions_snapshot()
        })

    def _create_commission_model(self):
        """创建佣金模型"""
        market = self.config.get("market", "a_share")

        if market == "a_share":
            return AShareCommission()
        elif market == "hk_stock":
            return HKStockCommission()
        elif market == "us_stock":
            return USStockCommission()

    def _create_slippage_model(self):
        """创建滑点模型"""
        slippage_type = self.config.get("slippage_type", "fixed")

        if slippage_type == "fixed":
            return FixedSlippage(rate=self.config.get("slippage_rate", 0.001))
        elif slippage_type == "volume":
            return VolumeSlippage()

    def _calculate_metrics(self) -> Dict:
        """计算绩效指标"""
        calculator = MetricsCalculator(self.daily_records, self.trade_records)

        return {
            "returns": calculator.calculate_returns(),
            "risk": calculator.calculate_risk(),
            "trades": calculator.calculate_trade_stats(),
            "benchmark": calculator.compare_with_benchmark(
                self.config.get("benchmark", "000300.SH")  # 沪深300
            )
        }

    def _generate_report(self, metrics: Dict) -> str:
        """生成回测报告"""
        generator = ReportGenerator()
        return generator.generate(
            daily_records=self.daily_records,
            trade_records=self.trade_records,
            metrics=metrics,
            config=self.config
        )
```

### 3.2 策略基类

```python
from abc import ABC, abstractmethod

class BaseStrategy(ABC):
    """策略基类"""

    def __init__(self, name: str, params: Dict):
        self.name = name
        self.params = params
        self.context = None

    def initialize(self, context: 'BacktestContext'):
        """初始化策略"""
        self.context = context
        self.on_init()

    @abstractmethod
    def on_init(self):
        """策略初始化回调（子类实现）"""
        pass

    @abstractmethod
    def generate_signals(
        self,
        data: Dict[str, pd.DataFrame],
        date: datetime
    ) -> List[Dict]:
        """生成交易信号（子类实现）"""
        pass

    @abstractmethod
    def get_required_indicators(self) -> Dict[str, Dict]:
        """获取所需指标（子类实现）"""
        pass

    def get_position(self, symbol: str) -> Optional['Position']:
        """获取持仓"""
        return self.context.portfolio.get_position(symbol)

    def get_latest_price(self, data: Dict[str, pd.DataFrame], symbol: str) -> float:
        """获取最新价格"""
        return data[symbol].iloc[-1]["close"]


# 内置策略示例：均线交叉
class MACrossStrategy(BaseStrategy):
    """均线交叉策略"""

    def __init__(
        self,
        short_period: int = 5,
        long_period: int = 20,
        position_size: float = 0.3
    ):
        params = {
            "short_period": short_period,
            "long_period": long_period,
            "position_size": position_size
        }
        super().__init__("ma_cross", params)

    def on_init(self):
        logger.info(f"Initialized MA Cross Strategy: {self.params}")

    def get_required_indicators(self) -> Dict[str, Dict]:
        return {
            "ma": {"period": self.params["short_period"]},
            "ma": {"period": self.params["long_period"]}
        }

    def generate_signals(
        self,
        data: Dict[str, pd.DataFrame],
        date: datetime
    ) -> List[Dict]:
        signals = []

        for symbol, df in data.items():
            if len(df) < self.params["long_period"]:
                continue

            # 计算均线
            short_ma = df["close"].rolling(window=self.params["short_period"]).mean()
            long_ma = df["close"].rolling(window=self.params["long_period"]).mean()

            # 当前和前一天的均线值
            current_short = short_ma.iloc[-1]
            current_long = long_ma.iloc[-1]
            prev_short = short_ma.iloc[-2]
            prev_long = long_ma.iloc[-2]

            # 金叉：做多
            if prev_short <= prev_long and current_short > current_long:
                position = self.get_position(symbol)

                if position is None or position.quantity == 0:
                    # 计算买入数量
                    cash_available = self.context.portfolio.cash * self.params["position_size"]
                    current_price = df.iloc[-1]["close"]
                    quantity = int(cash_available / current_price / 100) * 100  # A股100股为1手

                    if quantity > 0:
                        signals.append({
                            "symbol": symbol,
                            "action": "BUY",
                            "quantity": quantity,
                            "reason": f"MA{self.params['short_period']} golden cross MA{self.params['long_period']}"
                        })

            # 死叉：平仓
            elif prev_short >= prev_long and current_short < current_long:
                position = self.get_position(symbol)

                if position and position.quantity > 0:
                    signals.append({
                        "symbol": symbol,
                        "action": "SELL",
                        "quantity": position.quantity,
                        "reason": f"MA{self.params['short_period']} death cross MA{self.params['long_period']}"
                    })

        return signals
```

### 3.3 投资组合管理

```python
from dataclasses import dataclass
from typing import Dict, List, Optional

@dataclass
class Position:
    """持仓"""
    symbol: str
    quantity: int
    avg_price: float
    current_price: float
    market_value: float
    cost: float
    profit_loss: float
    profit_loss_pct: float

@dataclass
class Order:
    """订单"""
    order_id: str
    symbol: str
    quantity: int
    order_type: str  # market, limit
    direction: str   # long, short
    price: Optional[float]
    status: str      # pending, executed, cancelled
    created_at: datetime
    executed_at: Optional[datetime]

@dataclass
class Trade:
    """成交记录"""
    trade_id: str
    order_id: str
    symbol: str
    direction: str
    quantity: int
    price: float
    commission: float
    slippage: float
    amount: float
    executed_at: datetime

class Portfolio:
    """投资组合"""

    def __init__(
        self,
        initial_capital: float,
        commission_model,
        slippage_model
    ):
        self.initial_capital = initial_capital
        self.cash = initial_capital
        self.commission_model = commission_model
        self.slippage_model = slippage_model

        # 持仓
        self.positions: Dict[str, Position] = {}

        # 订单
        self.pending_orders: List[Order] = []
        self.executed_orders: List[Order] = []

        # 统计
        self.position_value = 0.0
        self.total_value = initial_capital
        self.returns = 0.0

    def create_order(
        self,
        symbol: str,
        quantity: int,
        order_type: str = "market",
        direction: str = "long",
        price: Optional[float] = None
    ) -> Order:
        """创建订单"""
        order = Order(
            order_id=self._generate_order_id(),
            symbol=symbol,
            quantity=quantity,
            order_type=order_type,
            direction=direction,
            price=price,
            status="pending",
            created_at=datetime.now(),
            executed_at=None
        )

        self.pending_orders.append(order)
        return order

    def execute_order(
        self,
        order: Order,
        execution_price: float,
        execution_date: datetime
    ) -> Trade:
        """执行订单"""
        # 计算佣金
        commission = self.commission_model.calculate(
            price=execution_price,
            quantity=order.quantity,
            direction=order.direction
        )

        # 计算交易金额
        amount = execution_price * order.quantity

        # 创建成交记录
        trade = Trade(
            trade_id=self._generate_trade_id(),
            order_id=order.order_id,
            symbol=order.symbol,
            direction=order.direction,
            quantity=order.quantity,
            price=execution_price,
            commission=commission,
            slippage=0.0,  # 滑点已包含在 execution_price 中
            amount=amount,
            executed_at=execution_date
        )

        # 更新持仓和资金
        if order.direction == "long":
            self._process_buy(trade)
        else:
            self._process_sell(trade)

        # 更新订单状态
        order.status = "executed"
        order.executed_at = execution_date
        self.executed_orders.append(order)

        return trade

    def _process_buy(self, trade: Trade):
        """处理买入"""
        total_cost = trade.amount + trade.commission

        # 检查资金是否充足
        if total_cost > self.cash:
            raise ValueError(f"Insufficient cash: need {total_cost}, have {self.cash}")

        # 扣除资金
        self.cash -= total_cost

        # 更新持仓
        if trade.symbol in self.positions:
            pos = self.positions[trade.symbol]
            # 更新平均成本
            total_quantity = pos.quantity + trade.quantity
            pos.avg_price = (pos.avg_price * pos.quantity + trade.price * trade.quantity) / total_quantity
            pos.quantity = total_quantity
            pos.cost = pos.avg_price * pos.quantity
        else:
            self.positions[trade.symbol] = Position(
                symbol=trade.symbol,
                quantity=trade.quantity,
                avg_price=trade.price,
                current_price=trade.price,
                market_value=trade.amount,
                cost=trade.amount,
                profit_loss=0.0,
                profit_loss_pct=0.0
            )

    def _process_sell(self, trade: Trade):
        """处理卖出"""
        if trade.symbol not in self.positions:
            raise ValueError(f"No position for {trade.symbol}")

        pos = self.positions[trade.symbol]

        if trade.quantity > pos.quantity:
            raise ValueError(f"Insufficient position: need {trade.quantity}, have {pos.quantity}")

        # 增加资金
        self.cash += trade.amount - trade.commission

        # 更新持仓
        pos.quantity -= trade.quantity

        if pos.quantity == 0:
            # 清空持仓
            del self.positions[trade.symbol]
        else:
            # 保持平均成本不变
            pos.cost = pos.avg_price * pos.quantity

    def update_market_value(self, data: Dict[str, pd.DataFrame], date: datetime):
        """更新持仓市值"""
        self.position_value = 0.0

        for symbol, pos in self.positions.items():
            if symbol in data and date in data[symbol].index:
                current_price = data[symbol].loc[date, "close"]
                pos.current_price = current_price
                pos.market_value = current_price * pos.quantity
                pos.profit_loss = pos.market_value - pos.cost
                pos.profit_loss_pct = pos.profit_loss / pos.cost if pos.cost > 0 else 0.0

                self.position_value += pos.market_value

        self.total_value = self.cash + self.position_value
        self.returns = (self.total_value - self.initial_capital) / self.initial_capital

    def get_position(self, symbol: str) -> Optional[Position]:
        """获取持仓"""
        return self.positions.get(symbol)

    def get_positions_snapshot(self) -> Dict:
        """获取持仓快照"""
        return {
            symbol: {
                "quantity": pos.quantity,
                "avg_price": pos.avg_price,
                "current_price": pos.current_price,
                "market_value": pos.market_value,
                "profit_loss": pos.profit_loss,
                "profit_loss_pct": pos.profit_loss_pct
            }
            for symbol, pos in self.positions.items()
        }

    def clear_executed_orders(self):
        """清理已执行订单"""
        self.pending_orders = [o for o in self.pending_orders if o.status == "pending"]

    def _generate_order_id(self) -> str:
        import uuid
        return f"O{uuid.uuid4().hex[:8]}"

    def _generate_trade_id(self) -> str:
        import uuid
        return f"T{uuid.uuid4().hex[:8]}"
```

### 3.4 佣金和滑点模型

```python
class AShareCommission:
    """A股佣金模型"""

    def __init__(
        self,
        commission_rate: float = 0.00025,  # 万2.5
        min_commission: float = 5.0,        # 最低5元
        stamp_tax_rate: float = 0.001       # 千1印花税（仅卖出）
    ):
        self.commission_rate = commission_rate
        self.min_commission = min_commission
        self.stamp_tax_rate = stamp_tax_rate

    def calculate(self, price: float, quantity: int, direction: str) -> float:
        """计算佣金"""
        amount = price * quantity

        # 佣金
        commission = max(amount * self.commission_rate, self.min_commission)

        # 印花税（仅卖出）
        if direction == "short":
            stamp_tax = amount * self.stamp_tax_rate
            commission += stamp_tax

        return commission


class FixedSlippage:
    """固定滑点模型"""

    def __init__(self, rate: float = 0.001):
        self.rate = rate

    def apply(self, price: float, direction: str, volume: int) -> float:
        """应用滑点"""
        if direction == "long":
            return price * (1 + self.rate)
        else:
            return price * (1 - self.rate)


class VolumeSlippage:
    """基于成交量的动态滑点"""

    def __init__(self, base_rate: float = 0.0005):
        self.base_rate = base_rate

    def apply(self, price: float, direction: str, volume: int) -> float:
        """应用滑点（简化版）"""
        # 实际应该根据订单量占市场成交量比例动态调整
        slippage_rate = self.base_rate * (1 + volume / 100000)
        slippage_rate = min(slippage_rate, 0.01)  # 最大1%

        if direction == "long":
            return price * (1 + slippage_rate)
        else:
            return price * (1 - slippage_rate)
```

## 4. 绩效指标计算

```python
class MetricsCalculator:
    """绩效指标计算器"""

    def __init__(self, daily_records: List[Dict], trade_records: List[Trade]):
        self.daily_records = pd.DataFrame(daily_records)
        self.trade_records = trade_records

    def calculate_returns(self) -> Dict:
        """计算收益指标"""
        total_return = self.daily_records["returns"].iloc[-1]
        daily_returns = self.daily_records["total_value"].pct_change().dropna()

        # 年化收益率
        trading_days = len(self.daily_records)
        years = trading_days / 252
        annual_return = (1 + total_return) ** (1 / years) - 1 if years > 0 else 0

        return {
            "total_return": total_return,
            "annual_return": annual_return,
            "daily_returns": daily_returns.tolist()
        }

    def calculate_risk(self) -> Dict:
        """计算风险指标"""
        daily_returns = self.daily_records["total_value"].pct_change().dropna()

        # 最大回撤
        cumulative = (1 + daily_returns).cumprod()
        running_max = cumulative.cummax()
        drawdown = (cumulative - running_max) / running_max
        max_drawdown = drawdown.min()

        # 波动率（年化）
        volatility = daily_returns.std() * np.sqrt(252)

        # 夏普比率（假设无风险利率3%）
        risk_free_rate = 0.03
        annual_return = self.calculate_returns()["annual_return"]
        sharpe_ratio = (annual_return - risk_free_rate) / volatility if volatility > 0 else 0

        # 索提诺比率
        downside_returns = daily_returns[daily_returns < 0]
        downside_deviation = downside_returns.std() * np.sqrt(252)
        sortino_ratio = (annual_return - risk_free_rate) / downside_deviation if downside_deviation > 0 else 0

        return {
            "max_drawdown": max_drawdown,
            "volatility": volatility,
            "sharpe_ratio": sharpe_ratio,
            "sortino_ratio": sortino_ratio
        }

    def calculate_trade_stats(self) -> Dict:
        """计算交易统计"""
        if not self.trade_records:
            return {}

        # 计算每笔交易盈亏（简化：配对买卖）
        buy_trades = [t for t in self.trade_records if t.direction == "long"]
        sell_trades = [t for t in self.trade_records if t.direction == "short"]

        wins = 0
        losses = 0
        total_profit = 0
        total_loss = 0

        for buy, sell in zip(buy_trades, sell_trades):
            if buy.symbol == sell.symbol:
                pnl = (sell.price - buy.price) * buy.quantity - buy.commission - sell.commission

                if pnl > 0:
                    wins += 1
                    total_profit += pnl
                else:
                    losses += 1
                    total_loss += abs(pnl)

        total_trades = wins + losses
        win_rate = wins / total_trades if total_trades > 0 else 0
        profit_loss_ratio = (total_profit / wins) / (total_loss / losses) if wins > 0 and losses > 0 else 0

        return {
            "total_trades": total_trades,
            "wins": wins,
            "losses": losses,
            "win_rate": win_rate,
            "profit_loss_ratio": profit_loss_ratio,
            "avg_profit": total_profit / wins if wins > 0 else 0,
            "avg_loss": total_loss / losses if losses > 0 else 0
        }

    def compare_with_benchmark(self, benchmark_symbol: str) -> Dict:
        """与基准对比"""
        # 获取基准数据
        # 简化实现，实际需要从数据引擎获取
        return {
            "alpha": 0.0,
            "beta": 1.0,
            "correlation": 0.0
        }
```

## 8. 总结

回测引擎设计要点:
1. **真实模拟**: 考虑滑点、佣金、涨跌停等真实交易限制
2. **避免未来函数**: 严格使用历史数据,防止数据泄漏
3. **完整评估**: 提供全面的绩效和风险指标
4. **可扩展**: 支持自定义策略和评估指标
5. **向量化**: 使用向量化计算提高回测速度

回测引擎为策略验证提供了坚实基础,帮助识别有效策略并优化参数。

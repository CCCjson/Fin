# 分析引擎设计文档

## 1. 模块概述

### 1.1 核心职责
分析引擎是量化交易系统的"大脑",负责:
- 计算各类技术指标（趋势、震荡、波动率、成交量）
- 检测买卖信号
- 多股票批量筛选
- K线形态识别
- 信号强度评分

### 1.2 设计目标
- **准确性**: 指标计算精确,符合金融标准
- **性能**: 支持向量化批量计算
- **可扩展**: 易于添加自定义指标和策略
- **可配置**: 参数灵活可调
- **可解释**: 信号原因清晰可追溯

## 3. 技术指标设计

### 3.1 指标基类

```python
from abc import ABC, abstractmethod
from typing import Dict, Any, Optional
import pandas as pd
import numpy as np

class BaseIndicator(ABC):
    """技术指标抽象基类"""

    def __init__(self, name: str, params: Dict[str, Any]):
        self.name = name
        self.params = params
        self._validate_params()

    @abstractmethod
    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        """计算指标

        Args:
            df: 包含 OHLCV 数据的 DataFrame

        Returns:
            添加了指标列的 DataFrame
        """
        pass

    @abstractmethod
    def _validate_params(self):
        """验证参数有效性"""
        pass

    def get_required_columns(self) -> list:
        """获取所需的输入列"""
        return ["close"]

    def get_output_columns(self) -> list:
        """获取输出的指标列名"""
        return [self.name]

    def __repr__(self):
        return f"{self.__class__.__name__}({self.params})"
```

### 3.2 趋势指标 (Trend Indicators)

```python
class MovingAverage(BaseIndicator):
    """移动平均线 (MA/EMA)"""

    def __init__(
        self,
        period: int = 20,
        ma_type: str = "sma",  # sma, ema, wma
        column: str = "close"
    ):
        params = {"period": period, "ma_type": ma_type, "column": column}
        super().__init__(f"ma_{period}", params)

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")
        if self.params["ma_type"] not in ["sma", "ema", "wma"]:
            raise ValueError("Invalid ma_type")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        column = self.params["column"]
        period = self.params["period"]
        ma_type = self.params["ma_type"]

        if ma_type == "sma":
            df[self.name] = df[column].rolling(window=period).mean()
        elif ma_type == "ema":
            df[self.name] = df[column].ewm(span=period, adjust=False).mean()
        elif ma_type == "wma":
            weights = np.arange(1, period + 1)
            df[self.name] = df[column].rolling(window=period).apply(
                lambda x: np.dot(x, weights) / weights.sum(), raw=True
            )

        return df


class MACD(BaseIndicator):
    """MACD 指标"""

    def __init__(
        self,
        fast_period: int = 12,
        slow_period: int = 26,
        signal_period: int = 9
    ):
        params = {
            "fast_period": fast_period,
            "slow_period": slow_period,
            "signal_period": signal_period
        }
        super().__init__("macd", params)

    def _validate_params(self):
        if self.params["fast_period"] >= self.params["slow_period"]:
            raise ValueError("fast_period must be < slow_period")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        fast = self.params["fast_period"]
        slow = self.params["slow_period"]
        signal = self.params["signal_period"]

        # 计算 EMA
        ema_fast = df["close"].ewm(span=fast, adjust=False).mean()
        ema_slow = df["close"].ewm(span=slow, adjust=False).mean()

        # MACD = EMA_fast - EMA_slow
        df["macd_dif"] = ema_fast - ema_slow

        # Signal = EMA of MACD
        df["macd_dea"] = df["macd_dif"].ewm(span=signal, adjust=False).mean()

        # Histogram = MACD - Signal
        df["macd_hist"] = df["macd_dif"] - df["macd_dea"]

        return df

    def get_output_columns(self) -> list:
        return ["macd_dif", "macd_dea", "macd_hist"]


class ADX(BaseIndicator):
    """平均趋向指数 (Average Directional Index)"""

    def __init__(self, period: int = 14):
        super().__init__("adx", {"period": period})

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]

        # 计算 +DM 和 -DM
        high_diff = df["high"].diff()
        low_diff = -df["low"].diff()

        plus_dm = np.where((high_diff > low_diff) & (high_diff > 0), high_diff, 0)
        minus_dm = np.where((low_diff > high_diff) & (low_diff > 0), low_diff, 0)

        # 计算 TR (True Range)
        tr1 = df["high"] - df["low"]
        tr2 = abs(df["high"] - df["close"].shift())
        tr3 = abs(df["low"] - df["close"].shift())
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)

        # 平滑处理
        atr = tr.rolling(window=period).mean()
        plus_di = 100 * pd.Series(plus_dm).rolling(window=period).mean() / atr
        minus_di = 100 * pd.Series(minus_dm).rolling(window=period).mean() / atr

        # 计算 DX
        dx = 100 * abs(plus_di - minus_di) / (plus_di + minus_di)

        # 计算 ADX
        df["adx"] = dx.rolling(window=period).mean()
        df["plus_di"] = plus_di
        df["minus_di"] = minus_di

        return df

    def get_output_columns(self) -> list:
        return ["adx", "plus_di", "minus_di"]
```

### 3.3 震荡指标 (Oscillator Indicators)

```python
class RSI(BaseIndicator):
    """相对强弱指数"""

    def __init__(self, period: int = 14):
        super().__init__("rsi", {"period": period})

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]

        # 计算价格变化
        delta = df["close"].diff()

        # 分离涨跌
        gain = delta.where(delta > 0, 0)
        loss = -delta.where(delta < 0, 0)

        # 计算平均涨跌幅
        avg_gain = gain.rolling(window=period).mean()
        avg_loss = loss.rolling(window=period).mean()

        # 计算 RS 和 RSI
        rs = avg_gain / avg_loss
        df["rsi"] = 100 - (100 / (1 + rs))

        return df


class KDJ(BaseIndicator):
    """KDJ 随机指标"""

    def __init__(
        self,
        fastk_period: int = 9,
        slowk_period: int = 3,
        slowd_period: int = 3
    ):
        params = {
            "fastk_period": fastk_period,
            "slowk_period": slowk_period,
            "slowd_period": slowd_period
        }
        super().__init__("kdj", params)

    def _validate_params(self):
        pass

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        fastk_period = self.params["fastk_period"]
        slowk_period = self.params["slowk_period"]
        slowd_period = self.params["slowd_period"]

        # 计算 RSV (Raw Stochastic Value)
        low_min = df["low"].rolling(window=fastk_period).min()
        high_max = df["high"].rolling(window=fastk_period).max()
        rsv = 100 * (df["close"] - low_min) / (high_max - low_min)

        # 计算 K 值（RSV 的移动平均）
        df["kdj_k"] = rsv.ewm(span=slowk_period, adjust=False).mean()

        # 计算 D 值（K 的移动平均）
        df["kdj_d"] = df["kdj_k"].ewm(span=slowd_period, adjust=False).mean()

        # 计算 J 值
        df["kdj_j"] = 3 * df["kdj_k"] - 2 * df["kdj_d"]

        return df

    def get_output_columns(self) -> list:
        return ["kdj_k", "kdj_d", "kdj_j"]


class CCI(BaseIndicator):
    """顺势指标 (Commodity Channel Index)"""

    def __init__(self, period: int = 20):
        super().__init__("cci", {"period": period})

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]

        # 计算典型价格 (Typical Price)
        tp = (df["high"] + df["low"] + df["close"]) / 3

        # 计算 SMA 和 Mean Deviation
        sma_tp = tp.rolling(window=period).mean()
        mad = tp.rolling(window=period).apply(lambda x: np.abs(x - x.mean()).mean())

        # 计算 CCI
        df["cci"] = (tp - sma_tp) / (0.015 * mad)

        return df
```

### 3.4 波动率指标 (Volatility Indicators)

```python
class BollingerBands(BaseIndicator):
    """布林带"""

    def __init__(self, period: int = 20, std_dev: float = 2.0):
        params = {"period": period, "std_dev": std_dev}
        super().__init__("boll", params)

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")
        if self.params["std_dev"] <= 0:
            raise ValueError("std_dev must be > 0")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]
        std_dev = self.params["std_dev"]

        # 中轨（移动平均）
        df["boll_mid"] = df["close"].rolling(window=period).mean()

        # 标准差
        rolling_std = df["close"].rolling(window=period).std()

        # 上轨和下轨
        df["boll_upper"] = df["boll_mid"] + std_dev * rolling_std
        df["boll_lower"] = df["boll_mid"] - std_dev * rolling_std

        # 计算带宽
        df["boll_width"] = (df["boll_upper"] - df["boll_lower"]) / df["boll_mid"]

        # 计算 %B (价格在带中的位置)
        df["boll_pct_b"] = (df["close"] - df["boll_lower"]) / (df["boll_upper"] - df["boll_lower"])

        return df

    def get_output_columns(self) -> list:
        return ["boll_upper", "boll_mid", "boll_lower", "boll_width", "boll_pct_b"]


class ATR(BaseIndicator):
    """平均真实波动幅度 (Average True Range)"""

    def __init__(self, period: int = 14):
        super().__init__("atr", {"period": period})

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]

        # 计算 True Range
        high_low = df["high"] - df["low"]
        high_close = abs(df["high"] - df["close"].shift())
        low_close = abs(df["low"] - df["close"].shift())

        tr = pd.concat([high_low, high_close, low_close], axis=1).max(axis=1)

        # 计算 ATR (Wilder's smoothing)
        df["atr"] = tr.ewm(alpha=1/period, adjust=False).mean()

        return df


class HistoricalVolatility(BaseIndicator):
    """历史波动率"""

    def __init__(self, period: int = 20, annualize: bool = True):
        params = {"period": period, "annualize": annualize}
        super().__init__("hv", params)

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]
        annualize = self.params["annualize"]

        # 计算对数收益率
        log_returns = np.log(df["close"] / df["close"].shift())

        # 计算滚动标准差
        df["hv"] = log_returns.rolling(window=period).std()

        # 年化
        if annualize:
            df["hv"] = df["hv"] * np.sqrt(252)  # 假设一年 252 个交易日

        return df
```

### 3.5 成交量指标 (Volume Indicators)

```python
class OBV(BaseIndicator):
    """能量潮 (On-Balance Volume)"""

    def __init__(self):
        super().__init__("obv", {})

    def _validate_params(self):
        pass

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        # 计算价格变化方向
        price_change = df["close"].diff()

        # 根据价格变化调整成交量符号
        volume_signed = df["volume"].copy()
        volume_signed[price_change < 0] = -volume_signed[price_change < 0]
        volume_signed[price_change == 0] = 0

        # 累计成交量
        df["obv"] = volume_signed.cumsum()

        return df


class VWAP(BaseIndicator):
    """成交量加权平均价格 (Volume Weighted Average Price)"""

    def __init__(self, period: Optional[int] = None):
        params = {"period": period}
        super().__init__("vwap", params)

    def _validate_params(self):
        pass

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]

        # 计算典型价格
        typical_price = (df["high"] + df["low"] + df["close"]) / 3

        # 计算 VWAP
        if period is None:
            # 当日累计 VWAP
            df["vwap"] = (typical_price * df["volume"]).cumsum() / df["volume"].cumsum()
        else:
            # 滚动 VWAP
            tp_volume = typical_price * df["volume"]
            df["vwap"] = tp_volume.rolling(window=period).sum() / df["volume"].rolling(window=period).sum()

        return df


class VolumeRatio(BaseIndicator):
    """量比"""

    def __init__(self, period: int = 5):
        super().__init__("volume_ratio", {"period": period})

    def _validate_params(self):
        if self.params["period"] < 2:
            raise ValueError("Period must be >= 2")

    def calculate(self, df: pd.DataFrame) -> pd.DataFrame:
        period = self.params["period"]

        # 计算平均成交量
        avg_volume = df["volume"].rolling(window=period).mean()

        # 量比 = 当前成交量 / 平均成交量
        df["volume_ratio"] = df["volume"] / avg_volume

        return df
```

### 3.6 指标注册器

class IndicatorRegistry:
    """指标注册器"""

    _indicators = {}

    @classmethod
    def register(cls, name: str, indicator_class: type):
        """注册指标"""
        cls._indicators[name] = indicator_class

    @classmethod
    def get(cls, name: str, **params) -> BaseIndicator:
        """获取指标实例"""
        if name not in cls._indicators:
            raise ValueError(f"Unknown indicator: {name}")
        return cls._indicators[name](**params)

    @classmethod
    def list_all(cls) -> list:
        """列出所有已注册指标"""
        return list(cls._indicators.keys())

## 4. 信号检测设计

### 4.1 交易规则定义

```python
from enum import Enum
from typing import Callable
from pydantic import BaseModel

class SignalType(Enum):
    """信号类型"""
    BUY = "BUY"
    SELL = "SELL"
    HOLD = "HOLD"

class SignalStrength(Enum):
    """信号强度"""
    WEAK = 0.3
    MODERATE = 0.6
    STRONG = 0.9

class TradingRule(BaseModel):
    """交易规则"""
    name: str
    description: str
    signal_type: SignalType
    condition: Callable[[pd.DataFrame], pd.Series]
    strength: float = SignalStrength.MODERATE.value
    params: dict = {}

    class Config:
        arbitrary_types_allowed = True


# 内置交易规则
class BuiltinRules:
    """内置交易规则库"""

    @staticmethod
    def ma_golden_cross(short_period: int = 5, long_period: int = 20) -> TradingRule:
        """均线金叉"""
        def condition(df: pd.DataFrame) -> pd.Series:
            ma_short = df["close"].rolling(window=short_period).mean()
            ma_long = df["close"].rolling(window=long_period).mean()

            # 金叉：短期均线上穿长期均线
            cross_up = (ma_short > ma_long) & (ma_short.shift() <= ma_long.shift())
            return cross_up

        return TradingRule(
            name="ma_golden_cross",
            description=f"MA{short_period} 上穿 MA{long_period}",
            signal_type=SignalType.BUY,
            condition=condition,
            strength=SignalStrength.MODERATE.value,
            params={"short_period": short_period, "long_period": long_period}
        )

    @staticmethod
    def ma_death_cross(short_period: int = 5, long_period: int = 20) -> TradingRule:
        """均线死叉"""
        def condition(df: pd.DataFrame) -> pd.Series:
            ma_short = df["close"].rolling(window=short_period).mean()
            ma_long = df["close"].rolling(window=long_period).mean()

            # 死叉：短期均线下穿长期均线
            cross_down = (ma_short < ma_long) & (ma_short.shift() >= ma_long.shift())
            return cross_down

        return TradingRule(
            name="ma_death_cross",
            description=f"MA{short_period} 下穿 MA{long_period}",
            signal_type=SignalType.SELL,
            condition=condition,
            strength=SignalStrength.MODERATE.value,
            params={"short_period": short_period, "long_period": long_period}
        )

    @staticmethod
    def macd_golden_cross() -> TradingRule:
        """MACD 金叉"""
        def condition(df: pd.DataFrame) -> pd.Series:
            # 需要先计算 MACD
            if "macd_dif" not in df.columns or "macd_dea" not in df.columns:
                macd = MACD()
                df = macd.calculate(df)

            # DIF 上穿 DEA
            cross_up = (df["macd_dif"] > df["macd_dea"]) & (df["macd_dif"].shift() <= df["macd_dea"].shift())
            return cross_up

        return TradingRule(
            name="macd_golden_cross",
            description="MACD DIF 上穿 DEA",
            signal_type=SignalType.BUY,
            condition=condition,
            strength=SignalStrength.STRONG.value
        )

    @staticmethod
    def rsi_oversold(threshold: float = 30) -> TradingRule:
        """RSI 超卖"""
        def condition(df: pd.DataFrame) -> pd.Series:
            if "rsi" not in df.columns:
                rsi = RSI()
                df = rsi.calculate(df)

            # RSI 从超卖区回升
            oversold = (df["rsi"] > threshold) & (df["rsi"].shift() <= threshold) & (df["rsi"].shift() < 50)
            return oversold

        return TradingRule(
            name="rsi_oversold",
            description=f"RSI 从 {threshold} 以下回升",
            signal_type=SignalType.BUY,
            condition=condition,
            strength=SignalStrength.MODERATE.value,
            params={"threshold": threshold}
        )

    @staticmethod
    def rsi_overbought(threshold: float = 70) -> TradingRule:
        """RSI 超买"""
        def condition(df: pd.DataFrame) -> pd.Series:
            if "rsi" not in df.columns:
                rsi = RSI()
                df = rsi.calculate(df)

            # RSI 从超买区回落
            overbought = (df["rsi"] < threshold) & (df["rsi"].shift() >= threshold) & (df["rsi"].shift() > 50)
            return overbought

        return TradingRule(
            name="rsi_overbought",
            description=f"RSI 从 {threshold} 以上回落",
            signal_type=SignalType.SELL,
            condition=condition,
            strength=SignalStrength.MODERATE.value,
            params={"threshold": threshold}
        )

    @staticmethod
    def boll_breakout_lower() -> TradingRule:
        """布林带下轨突破"""
        def condition(df: pd.DataFrame) -> pd.Series:
            if "boll_lower" not in df.columns:
                boll = BollingerBands()
                df = boll.calculate(df)

            # 价格触及下轨后反弹
            touch_lower = df["close"] <= df["boll_lower"]
            bounce_back = df["close"] > df["boll_lower"]
            signal = touch_lower.shift() & bounce_back
            return signal

        return TradingRule(
            name="boll_breakout_lower",
            description="价格触及布林带下轨后反弹",
            signal_type=SignalType.BUY,
            condition=condition,
            strength=SignalStrength.MODERATE.value
        )

    @staticmethod
    def volume_breakout(ratio: float = 2.0) -> TradingRule:
        """放量突破"""
        def condition(df: pd.DataFrame) -> pd.Series:
            avg_volume = df["volume"].rolling(window=20).mean()
            volume_surge = df["volume"] > ratio * avg_volume

            # 价格突破前高
            high_20 = df["high"].rolling(window=20).max()
            price_breakout = df["close"] > high_20.shift()

            signal = volume_surge & price_breakout
            return signal

        return TradingRule(
            name="volume_breakout",
            description=f"放量（{ratio}倍）突破前高",
            signal_type=SignalType.BUY,
            condition=condition,
            strength=SignalStrength.STRONG.value,
            params={"ratio": ratio}
        )

    @staticmethod
    def macd_divergence() -> TradingRule:
        """MACD 底背离"""
        def condition(df: pd.DataFrame) -> pd.Series:
            if "macd_dif" not in df.columns:
                macd = MACD()
                df = macd.calculate(df)

            # 找到价格和 MACD 的低点
            price_low = df["close"].rolling(window=5, center=True).min() == df["close"]
            macd_low = df["macd_dif"].rolling(window=5, center=True).min() == df["macd_dif"]

            # 检测背离（价格新低，MACD 未新低）
            # 简化实现，实际需要更复杂的逻辑
            divergence = price_low & macd_low
            return divergence

        return TradingRule(
            name="macd_divergence",
            description="MACD 底背离",
            signal_type=SignalType.BUY,
            condition=condition,
            strength=SignalStrength.STRONG.value
        )
```

### 4.2 信号检测器

```python
from typing import List, Dict
from datetime import datetime

class Signal(BaseModel):
    """信号模型"""
    symbol: str
    date: datetime
    signal_type: SignalType
    strength: float
    reasons: List[Dict[str, str]]
    price: float
    suggested_action: Dict[str, Any] = {}

class SignalDetector:
    """信号检测器"""

    def __init__(self, rules: List[TradingRule]):
        self.rules = rules

    def detect(self, df: pd.DataFrame, symbol: str) -> List[Signal]:
        """检测信号"""
        signals = []

        # 为每一行检测信号
        for idx in range(len(df)):
            if idx < max([rule.params.get("short_period", 0) for rule in self.rules if rule.params]):
                continue  # 跳过数据不足的行

            row_df = df.iloc[:idx+1]  # 只使用当前及之前的数据

            buy_reasons = []
            sell_reasons = []
            buy_strength = 0
            sell_strength = 0

            # 检查所有规则
            for rule in self.rules:
                try:
                    condition_result = rule.condition(row_df)

                    if condition_result.iloc[-1]:  # 最后一行满足条件
                        reason = {
                            "indicator": rule.name,
                            "detail": rule.description
                        }

                        if rule.signal_type == SignalType.BUY:
                            buy_reasons.append(reason)
                            buy_strength += rule.strength
                        elif rule.signal_type == SignalType.SELL:
                            sell_reasons.append(reason)
                            sell_strength += rule.strength

                except Exception as e:
                    logger.warning(f"Rule {rule.name} failed: {e}")

            # 生成信号
            current_row = df.iloc[idx]

            if buy_reasons:
                signals.append(Signal(
                    symbol=symbol,
                    date=current_row["date"],
                    signal_type=SignalType.BUY,
                    strength=min(buy_strength / len(buy_reasons), 1.0),
                    reasons=buy_reasons,
                    price=current_row["close"],
                    suggested_action=self._calculate_action(current_row, SignalType.BUY, df.iloc[:idx+1])
                ))

            if sell_reasons:
                signals.append(Signal(
                    symbol=symbol,
                    date=current_row["date"],
                    signal_type=SignalType.SELL,
                    strength=min(sell_strength / len(sell_reasons), 1.0),
                    reasons=sell_reasons,
                    price=current_row["close"],
                    suggested_action=self._calculate_action(current_row, SignalType.SELL, df.iloc[:idx+1])
                ))

        return signals

    def _calculate_action(
        self,
        row: pd.Series,
        signal_type: SignalType,
        df: pd.DataFrame
    ) -> Dict[str, Any]:
        """计算建议操作"""
        current_price = row["close"]

        if signal_type == SignalType.BUY:
            # 计算止损价（基于 ATR 或固定百分比）
            if "atr" in row:
                stop_loss = current_price - 2 * row["atr"]
            else:
                stop_loss = current_price * 0.95  # -5%

            # 计算止盈价
            risk = current_price - stop_loss
            take_profit = current_price + 2 * risk  # 风险回报比 1:2

            return {
                "entry_price": round(current_price, 2),
                "stop_loss": round(stop_loss, 2),
                "take_profit": round(take_profit, 2),
                "position_size": "5%"  # 建议仓位
            }
        else:
            return {
                "exit_price": round(current_price, 2)
            }
```

### 4.3 信号评分器

```python
class SignalEvaluator:
    """信号评分器"""

    @staticmethod
    def evaluate_signal_quality(signal: Signal, df: pd.DataFrame) -> float:
        """评估信号质量（0-1）"""
        score = signal.strength

        # 因素1：多信号共振（信号数量越多越好）
        num_reasons = len(signal.reasons)
        if num_reasons >= 3:
            score *= 1.2
        elif num_reasons == 1:
            score *= 0.8

        # 因素2：趋势一致性
        if "ma_20" in df.columns and "ma_60" in df.columns:
            ma20 = df["ma_20"].iloc[-1]
            ma60 = df["ma_60"].iloc[-1]

            if signal.signal_type == SignalType.BUY and ma20 > ma60:
                score *= 1.1
            elif signal.signal_type == SignalType.SELL and ma20 < ma60:
                score *= 1.1

        # 因素3：成交量确认
        if "volume_ratio" in df.columns:
            volume_ratio = df["volume_ratio"].iloc[-1]
            if volume_ratio > 1.5:  # 放量
                score *= 1.1

        return min(score, 1.0)

    @staticmethod
    def filter_signals(
        signals: List[Signal],
        min_strength: float = 0.6,
        min_reasons: int = 2
    ) -> List[Signal]:
        """过滤信号"""
        filtered = [
            s for s in signals
            if s.strength >= min_strength and len(s.reasons) >= min_reasons
        ]
        return filtered
```

### 4.4 股票筛选器

```python
class StockScreener:
    """股票筛选器"""

    def __init__(self, data_engine, analysis_engine):
        self.data_engine = data_engine
        self.analysis_engine = analysis_engine

    def screen(
        self,
        symbols: List[str],
        rules: List[TradingRule],
        filters: Dict[str, Any] = None
    ) -> List[Dict[str, Any]]:
        """批量筛选股票"""
        results = []

        for symbol in symbols:
            try:
                # 获取数据
                df = self.data_engine.get_daily_data(
                    symbol=symbol,
                    start_date=(datetime.now() - timedelta(days=365)).strftime("%Y-%m-%d"),
                    end_date=datetime.now().strftime("%Y-%m-%d")
                )

                # 计算指标
                df = self.analysis_engine.calculate_indicators(df, ["ma", "rsi", "macd", "boll"])

                # 检测信号
                detector = SignalDetector(rules)
                signals = detector.detect(df, symbol)

                # 只保留最近的信号
                recent_signals = [s for s in signals if (datetime.now() - s.date).days <= 5]

                if recent_signals:
                    results.append({
                        "symbol": symbol,
                        "signals": recent_signals,
                        "latest_price": df.iloc[-1]["close"],
                        "latest_volume": df.iloc[-1]["volume"]
                    })

            except Exception as e:
                logger.error(f"Failed to screen {symbol}: {e}")

        # 应用过滤器
        if filters:
            results = self._apply_filters(results, filters)

        # 按信号强度排序
        results.sort(key=lambda x: max(s.strength for s in x["signals"]), reverse=True)

        return results

    def _apply_filters(self, results: List[Dict], filters: Dict[str, Any]) -> List[Dict]:
        """应用过滤条件"""
        filtered = results

        # 价格过滤
        if "min_price" in filters:
            filtered = [r for r in filtered if r["latest_price"] >= filters["min_price"]]
        if "max_price" in filters:
            filtered = [r for r in filtered if r["latest_price"] <= filters["max_price"]]

        # 成交量过滤
        if "min_volume" in filters:
            filtered = [r for r in filtered if r["latest_volume"] >= filters["min_volume"]]

        # 信号强度过滤
        if "min_strength" in filters:
            filtered = [
                r for r in filtered
                if max(s.strength for s in r["signals"]) >= filters["min_strength"]
            ]

        return filtered
```

## 5. K线形态识别

```python
class CandlestickPatterns:
    """K线形态识别"""

    @staticmethod
    def is_hammer(row: pd.Series) -> bool:
        """锤子线"""
        body = abs(row["close"] - row["open"])
        upper_shadow = row["high"] - max(row["open"], row["close"])
        lower_shadow = min(row["open"], row["close"]) - row["low"]

        return (
            lower_shadow > 2 * body and
            upper_shadow < body
        )

    @staticmethod
    def is_engulfing(current: pd.Series, previous: pd.Series) -> bool:
        """吞没形态"""
        # 看涨吞没
        bullish_engulfing = (
            previous["close"] < previous["open"] and  # 前一根是阴线
            current["close"] > current["open"] and    # 当前是阳线
            current["open"] < previous["close"] and
            current["close"] > previous["open"]
        )

        # 看跌吞没
        bearish_engulfing = (
            previous["close"] > previous["open"] and
            current["close"] < current["open"] and
            current["open"] > previous["close"] and
            current["close"] < previous["open"]
        )

        return bullish_engulfing or bearish_engulfing

    @staticmethod
    def is_morning_star(df: pd.DataFrame, idx: int) -> bool:
        """早晨之星（需要三根K线）"""
        if idx < 2:
            return False

        first = df.iloc[idx-2]
        second = df.iloc[idx-1]
        third = df.iloc[idx]

        # 第一根：大阴线
        # 第二根：小实体（星线）
        # 第三根：大阳线

        first_bearish = first["close"] < first["open"]
        first_body = abs(first["close"] - first["open"])

        second_body = abs(second["close"] - second["open"])
        second_small = second_body < first_body * 0.3

        third_bullish = third["close"] > third["open"]
        third_body = abs(third["close"] - third["open"])

        return (
            first_bearish and
            second_small and
            third_bullish and
            third["close"] > (first["open"] + first["close"]) / 2
        )
```

```python
class AnalysisEngine:
    """分析引擎主类"""

    def __init__(self, config: dict):
        self.config = config
        self.indicator_registry = IndicatorRegistry()

    def calculate_indicators(
        self,
        df: pd.DataFrame,
        indicators: List[str],
        params: Dict[str, dict] = None
    ) -> pd.DataFrame:
        """批量计算指标"""
        params = params or {}

        for indicator_name in indicators:
            try:
                indicator_params = params.get(indicator_name, {})
                indicator = self.indicator_registry.get(indicator_name, **indicator_params)
                df = indicator.calculate(df)
            except Exception as e:
                logger.error(f"Failed to calculate {indicator_name}: {e}")

        return df

    def detect_signals(
        self,
        df: pd.DataFrame,
        symbol: str,
        strategy: str = "default"
    ) -> List[Signal]:
        """检测交易信号"""
        rules = self._get_strategy_rules(strategy)
        detector = SignalDetector(rules)
        return detector.detect(df, symbol)

    def _get_strategy_rules(self, strategy: str) -> List[TradingRule]:
        """获取策略规则"""
        if strategy == "default":
            return [
                BuiltinRules.ma_golden_cross(5, 20),
                BuiltinRules.macd_golden_cross(),
                BuiltinRules.rsi_oversold(30),
                BuiltinRules.boll_breakout_lower(),
                BuiltinRules.volume_breakout(2.0)
            ]
        # 可以扩展其他策略
        return []
```

# 数据引擎设计文档

## 1. 模块概述

### 1.1 核心职责
数据引擎是整个量化交易系统的基石，负责：
- 从多个数据源获取市场行情数据
- 数据标准化与质量控制
- 数据持久化存储
- 数据查询与缓存优化
- 定时任务调度与增量更新

### 1.2 设计目标
- **统一接口**：屏蔽不同数据源的差异
- **高可用**：多数据源冗余，自动故障切换
- **数据质量**：完整性校验、异常值检测
- **性能优化**：批量获取、增量更新、智能缓存

### 3.1 数据获取器 (Fetchers)

#### 3.1.1 抽象基类设计

```python
from abc import ABC, abstractmethod
from typing import Optional, List
from datetime import datetime
import pandas as pd
from pydantic import BaseModel

class MarketDataRequest(BaseModel):
    """数据请求模型"""
    symbol: str                      # 股票代码
    start_date: datetime             # 开始日期
    end_date: datetime               # 结束日期
    frequency: str = "1d"            # 频率: 1d, 1w, 1m
    adjust: str = "qfq"              # 复权方式: qfq(前复权), hfq(后复权), none

class MarketDataResponse(BaseModel):
    """数据响应模型"""
    symbol: str
    market: str                       # a_share, hk_stock, us_stock
    data: pd.DataFrame                # OHLCV 数据
    metadata: dict                    # 元数据（数据源、获取时间等）

class BaseFetcher(ABC):
    """数据获取器抽象基类"""

    def __init__(self, config: dict):
        self.config = config
        self.retry_times = config.get("retry_times", 3)
        self.retry_delay = config.get("retry_delay", 5)
        self.timeout = config.get("timeout", 30)

    @abstractmethod
    def fetch_daily(self, request: MarketDataRequest) -> MarketDataResponse:
        """获取日线数据"""
        pass

    @abstractmethod
    def fetch_realtime(self, symbols: List[str]) -> List[dict]:
        """获取实时数据"""
        pass

    @abstractmethod
    def validate_symbol(self, symbol: str) -> bool:
        """验证股票代码有效性"""
        pass

    @abstractmethod
    def search_symbol(self, keyword: str) -> List[dict]:
        """搜索股票"""
        pass

    def _retry_on_failure(self, func, *args, **kwargs):
        """失败重试装饰器"""
        import time
        from tenacity import retry, stop_after_attempt, wait_exponential

        @retry(
            stop=stop_after_attempt(self.retry_times),
            wait=wait_exponential(multiplier=1, min=2, max=10)
        )
        def wrapper():
            return func(*args, **kwargs)

        return wrapper()
```

```python
class AShareFetcher(BaseFetcher):
    """A股数据获取器 - 基于 akshare"""

    def __init__(self, config: dict):
        super().__init__(config)
        self.primary_source = "akshare"
        self.fallback_source = "tushare"

    def fetch_daily(self, request: MarketDataRequest) -> MarketDataResponse:
        """获取日线数据"""
        try:
            return self._fetch_from_akshare(request)
        except Exception as e:
            logger.warning(f"akshare failed: {e}, trying fallback...")
            return self._fetch_from_tushare(request)

    def _fetch_from_akshare(self, request: MarketDataRequest) -> MarketDataResponse:
        import akshare as ak

        # akshare 使用 6 位代码
        symbol_code = request.symbol[:6]

        # 获取数据
        df = ak.stock_zh_a_hist(
            symbol=symbol_code,
            period="daily",
            start_date=request.start_date.strftime("%Y%m%d"),
            end_date=request.end_date.strftime("%Y%m%d"),
            adjust=request.adjust
        )

        # 字段映射
        df = df.rename(columns={
            "日期": "date",
            "开盘": "open",
            "收盘": "close",
            "最高": "high",
            "最低": "low",
            "成交量": "volume",
            "成交额": "amount",
            "换手率": "turnover"
        })

        return MarketDataResponse(
            symbol=request.symbol,
            market="a_share",
            data=df,
            metadata={
                "source": "akshare",
                "fetched_at": datetime.now().isoformat(),
                "adjust": request.adjust
            }
        )

    def fetch_realtime(self, symbols: List[str]) -> List[dict]:
        """获取实时行情"""
        import akshare as ak

        result = []
        for symbol in symbols:
            try:
                df = ak.stock_zh_a_spot_em()
                stock = df[df["代码"] == symbol[:6]].iloc[0]

                result.append({
                    "symbol": symbol,
                    "price": float(stock["最新价"]),
                    "change": float(stock["涨跌幅"]),
                    "volume": int(stock["成交量"]),
                    "amount": float(stock["成交额"]),
                    "timestamp": datetime.now().isoformat()
                })
            except Exception as e:
                logger.error(f"Failed to fetch realtime data for {symbol}: {e}")

        return result

    def validate_symbol(self, symbol: str) -> bool:
        """验证股票代码"""
        # A股代码格式：6位数字 + 市场后缀（.SH/.SZ）
        import re
        pattern = r"^\d{6}\.(SH|SZ)$"
        return bool(re.match(pattern, symbol))

    def search_symbol(self, keyword: str) -> List[dict]:
        """搜索股票"""
        import akshare as ak

        df = ak.stock_zh_a_spot_em()
        # 支持代码或名称搜索
        matched = df[
            df["代码"].str.contains(keyword) |
            df["名称"].str.contains(keyword)
        ]

        return [
            {
                "symbol": f"{row['代码']}.SH" if row['代码'].startswith('6') else f"{row['代码']}.SZ",
                "name": row["名称"],
                "market": "a_share"
            }
            for _, row in matched.iterrows()
        ]
```

```python
class FetcherFactory:

        "a_share": AShareFetcher,
        "hk_stock": HKStockFetcher,

        if not fetcher_class:
            raise ValueError(f"Unknown market: {market}")

    def get_fetcher_for_symbol(cls, symbol: str, config: dict) -> BaseFetcher:

        if symbol.endswith(('.SH', '.SZ')):
            return cls.create("a_share", config)
        elif symbol.endswith('.HK'):
            return cls.create("hk_stock", config)
        else:
            return cls.create("us_stock", config)
```

#### 3.2.2 数据访问层 (Repository)

```python
from typing import List, Optional
from datetime import datetime, timedelta
from sqlalchemy.orm import Session
from sqlalchemy import and_, func

class QuoteRepository:
    """行情数据仓库"""

    def __init__(self, session: Session):
        self.session = session

    def save_daily_quotes(self, quotes: List[DailyQuote]) -> None:
        """批量保存日线数据（去重）"""
        for quote in quotes:
            existing = self.session.query(DailyQuote).filter(
                and_(
                    DailyQuote.symbol == quote.symbol,
                    DailyQuote.date == quote.date
                )
            ).first()

            if existing:
                # 更新现有记录
                for key, value in quote.__dict__.items():
                    if not key.startswith('_'):
                        setattr(existing, key, value)
            else:
                # 插入新记录
                self.session.add(quote)

        self.session.commit()

    def get_daily_quotes(
        self,
        symbol: str,
        start_date: datetime,
        end_date: datetime
    ) -> List[DailyQuote]:
        """查询日线数据"""
        return self.session.query(DailyQuote).filter(
            and_(
                DailyQuote.symbol == symbol,
                DailyQuote.date >= start_date,
                DailyQuote.date <= end_date
            )
        ).order_by(DailyQuote.date).all()

    def get_latest_date(self, symbol: str) -> Optional[datetime]:
        """获取最新数据日期（用于增量更新）"""
        result = self.session.query(func.max(DailyQuote.date)).filter(
            DailyQuote.symbol == symbol
        ).scalar()
        return result

    def get_symbols_by_market(self, market: str) -> List[str]:
        """获取市场所有股票代码"""
        return [
            row[0] for row in self.session.query(StockInfo.symbol).filter(
                StockInfo.market == market,
                StockInfo.is_active == 1
            ).all()
        ]

    def get_missing_data_ranges(
        self,
        symbol: str,
        start_date: datetime,
        end_date: datetime
    ) -> List[tuple]:
        """检测缺失数据的日期范围"""
        existing_dates = set([
            row[0] for row in self.session.query(DailyQuote.date).filter(
                and_(
                    DailyQuote.symbol == symbol,
                    DailyQuote.date >= start_date,
                    DailyQuote.date <= end_date
                )
            ).all()
        ])

        # 计算所有交易日（排除周末）
        all_dates = []
        current = start_date
        while current <= end_date:
            if current.weekday() < 5:  # 周一到周五
                all_dates.append(current)
            current += timedelta(days=1)

        # 找出缺失日期
        missing_dates = sorted(set(all_dates) - existing_dates)

        # 转换为连续范围
        ranges = []
        if missing_dates:
            start = missing_dates[0]
            end = missing_dates[0]

            for date in missing_dates[1:]:
                if (date - end).days <= 3:  # 允许周末间隔
                    end = date
                else:
                    ranges.append((start, end))
                    start = end = date

            ranges.append((start, end))

        return ranges
```

```python
class DataNormalizer:
    """数据标准化器"""

    @staticmethod
    def normalize_columns(df: pd.DataFrame) -> pd.DataFrame:
        """列名标准化"""
        column_mapping = {
            # 日期
            "日期": "date", "Date": "date", "trade_date": "date",
            # 开盘
            "开盘": "open", "Open": "open", "open_price": "open",
            # 最高
            "最高": "high", "High": "high", "high_price": "high",
            # 最低
            "最低": "low", "Low": "low", "low_price": "low",
            # 收盘
            "收盘": "close", "Close": "close", "close_price": "close",
            # 成交量
            "成交量": "volume", "Volume": "volume", "vol": "volume",
        }

        df = df.rename(columns=column_mapping)
        return df

    @staticmethod
    def normalize_datetime(df: pd.DataFrame, tz: str = "UTC") -> pd.DataFrame:
        """时间标准化为 UTC"""
        if "date" in df.columns:
            df["date"] = pd.to_datetime(df["date"])
            if df["date"].dt.tz is None:
                df["date"] = df["date"].dt.tz_localize(tz)
            else:
                df["date"] = df["date"].dt.tz_convert(tz)
        return df

    @staticmethod
    def normalize_numeric(df: pd.DataFrame) -> pd.DataFrame:
        """数值类型标准化"""
        numeric_columns = ["open", "high", "low", "close", "volume"]
        for col in numeric_columns:
            if col in df.columns:
                df[col] = pd.to_numeric(df[col], errors="coerce")
        return df

    @staticmethod
    def normalize(df: pd.DataFrame, market: str) -> pd.DataFrame:
        """完整标准化流程"""
        df = DataNormalizer.normalize_columns(df)
        df = DataNormalizer.normalize_datetime(df)
        df = DataNormalizer.normalize_numeric(df)

        # 确保必需列存在
        required_cols = ["date", "open", "high", "low", "close", "volume"]
        missing_cols = set(required_cols) - set(df.columns)
        if missing_cols:
            raise ValueError(f"Missing required columns: {missing_cols}")

        # 按日期排序
        df = df.sort_values("date").reset_index(drop=True)

        return df[required_cols + [col for col in df.columns if col not in required_cols]]
```

```python
class DataValidator:
    """数据验证器"""

    @staticmethod
    def validate_ohlc(df: pd.DataFrame) -> tuple[bool, List[str]]:
        """验证 OHLC 关系"""
        errors = []

        # 检查 high >= max(open, close, low)
        invalid_high = df[df["high"] < df[["open", "close", "low"]].max(axis=1)]
        if not invalid_high.empty:
            errors.append(f"Found {len(invalid_high)} rows where high < max(open, close, low)")

        # 检查 low <= min(open, close, high)
        invalid_low = df[df["low"] > df[["open", "close", "high"]].min(axis=1)]
        if not invalid_low.empty:
            errors.append(f"Found {len(invalid_low)} rows where low > min(open, close, high)")

        # 检查价格为正
        negative_prices = df[(df[["open", "high", "low", "close"]] <= 0).any(axis=1)]
        if not negative_prices.empty:
            errors.append(f"Found {len(negative_prices)} rows with non-positive prices")

        return len(errors) == 0, errors

    @staticmethod
    def validate_completeness(df: pd.DataFrame, start_date: datetime, end_date: datetime) -> tuple[bool, dict]:
        """验证数据完整性"""
        expected_days = (end_date - start_date).days + 1
        actual_days = len(df)

        # 计算交易日（排除周末）
        expected_trading_days = sum(
            1 for i in range(expected_days)
            if (start_date + timedelta(days=i)).weekday() < 5
        )

        completeness_ratio = actual_days / expected_trading_days if expected_trading_days > 0 else 0

        return completeness_ratio >= 0.95, {
            "expected_trading_days": expected_trading_days,
            "actual_days": actual_days,
            "completeness_ratio": completeness_ratio,
            "missing_days": expected_trading_days - actual_days
        }

    @staticmethod
    def validate_duplicates(df: pd.DataFrame) -> tuple[bool, int]:
        """检查重复数据"""
        duplicates = df.duplicated(subset=["date"]).sum()
        return duplicates == 0, duplicates
```

#### 3.3.3 数据清洗

```python
class DataCleaner:
    """数据清洗器"""

    @staticmethod
    def remove_duplicates(df: pd.DataFrame) -> pd.DataFrame:
        """去重（保留最新）"""
        return df.drop_duplicates(subset=["date"], keep="last")

    @staticmethod
    def fill_missing_values(df: pd.DataFrame, method: str = "ffill") -> pd.DataFrame:
        """填充缺失值"""
        if method == "ffill":
            # 前向填充
            df = df.fillna(method="ffill")
        elif method == "interpolate":
            # 线性插值
            df[["open", "high", "low", "close"]] = df[["open", "high", "low", "close"]].interpolate()
        elif method == "drop":
            # 删除缺失行
            df = df.dropna()

        return df

    @staticmethod
    def fix_outliers(df: pd.DataFrame, method: str = "iqr", threshold: float = 3.0) -> pd.DataFrame:
        """异常值处理"""
        if method == "iqr":
            # 使用 IQR 方法
            for col in ["open", "high", "low", "close"]:
                Q1 = df[col].quantile(0.25)
                Q3 = df[col].quantile(0.75)
                IQR = Q3 - Q1
                lower = Q1 - threshold * IQR
                upper = Q3 + threshold * IQR

                # 用边界值替换异常值
                df[col] = df[col].clip(lower, upper)

        elif method == "zscore":
            # 使用 Z-score 方法
            from scipy import stats
            for col in ["open", "high", "low", "close"]:
                z_scores = np.abs(stats.zscore(df[col]))
                df = df[z_scores < threshold]

        return df

    @staticmethod
    def clean(df: pd.DataFrame, config: dict = None) -> pd.DataFrame:
        """完整清洗流程"""
        config = config or {}

        # 1. 去重
        df = DataCleaner.remove_duplicates(df)

        # 2. 处理缺失值
        fill_method = config.get("fill_method", "ffill")
        df = DataCleaner.fill_missing_values(df, method=fill_method)

        # 3. 处理异常值
        if config.get("fix_outliers", False):
            df = DataCleaner.fix_outliers(df, method=config.get("outlier_method", "iqr"))

        return df
```

### 3.5 缓存管理

```python
from functools import lru_cache
import hashlib
import pickle

class DataCache:
    """数据缓存管理"""

    def __init__(self, max_size: int = 1000, ttl: int = 3600):
        self.max_size = max_size
        self.ttl = ttl  # 秒
        self._cache = {}
        self._timestamps = {}

    def _make_key(self, *args, **kwargs) -> str:
        """生成缓存键"""
        key_data = pickle.dumps((args, sorted(kwargs.items())))
        return hashlib.md5(key_data).hexdigest()

    def get(self, *args, **kwargs):
        """获取缓存"""
        key = self._make_key(*args, **kwargs)

        if key in self._cache:
            # 检查是否过期
            if time.time() - self._timestamps[key] < self.ttl:
                return self._cache[key]
            else:
                # 清理过期缓存
                del self._cache[key]
                del self._timestamps[key]

        return None

    def set(self, value, *args, **kwargs):
        """设置缓存"""
        key = self._make_key(*args, **kwargs)

        # LRU 淘汰
        if len(self._cache) >= self.max_size:
            oldest_key = min(self._timestamps, key=self._timestamps.get)
            del self._cache[oldest_key]
            del self._timestamps[oldest_key]

        self._cache[key] = value
        self._timestamps[key] = time.time()

    def clear(self):
        """清空缓存"""
        self._cache.clear()
        self._timestamps.clear()
```

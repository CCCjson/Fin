# 交易引擎设计文档

## 1. 模块概述

### 1.1 核心职责
交易引擎负责实际的交易执行和风险控制:
- 订单创建与管理
- 模拟交易撮合
- 实盘交易对接
- 持仓跟踪管理
- 风险控制检查
- 交易日志记录

### 1.2 设计目标
- **安全第一**: 多层风控,保护资金安全
- **精确可靠**: 订单状态精确跟踪
- **实时响应**: 快速处理交易指令
- **可审计**: 完整的交易日志
- **易切换**: 模拟与实盘无缝切换

### 1.3 交易模式
- **模拟交易**: 本地模拟撮合,用于策略验证
- **实盘交易**: 对接券商API,实际下单（富途/Alpaca等）

## 3. 核心组件设计

### 3.1 订单系统

```python
from enum import Enum
from dataclasses import dataclass
from typing import Optional
from datetime import datetime

class OrderType(Enum):
    """订单类型"""
    MARKET = "market"           # 市价单
    LIMIT = "limit"             # 限价单
    STOP = "stop"               # 止损单
    STOP_LIMIT = "stop_limit"   # 止损限价单

class OrderSide(Enum):
    """订单方向"""
    BUY = "buy"
    SELL = "sell"

class OrderStatus(Enum):
    """订单状态"""
    PENDING = "pending"             # 待提交
    SUBMITTED = "submitted"         # 已提交
    PARTIAL_FILLED = "partial_filled"  # 部分成交
    FILLED = "filled"               # 全部成交
    CANCELLED = "cancelled"         # 已取消
    REJECTED = "rejected"           # 被拒绝
    EXPIRED = "expired"             # 已过期

@dataclass
class Order:
    """订单模型"""
    order_id: str
    symbol: str
    side: OrderSide
    order_type: OrderType
    quantity: int
    price: Optional[float]          # 限价单价格
    stop_price: Optional[float]     # 止损价
    status: OrderStatus
    filled_quantity: int = 0
    avg_fill_price: float = 0.0
    commission: float = 0.0

    # 时间戳
    created_at: datetime = None
    submitted_at: Optional[datetime] = None
    filled_at: Optional[datetime] = None
    cancelled_at: Optional[datetime] = None

    # 元数据
    strategy: Optional[str] = None
    signal_strength: float = 0.0
    reason: str = ""

    # 风控
    risk_checked: bool = False
    rejected_reason: str = ""

class OrderManager:
    """订单管理器"""

    def __init__(self):
        self.orders: Dict[str, Order] = {}
        self.pending_orders: List[Order] = []
        self.active_orders: List[Order] = []
        self.completed_orders: List[Order] = []

    def create_order(
        self,
        symbol: str,
        side: OrderSide,
        quantity: int,
        order_type: OrderType = OrderType.MARKET,
        price: Optional[float] = None,
        stop_price: Optional[float] = None,
        **kwargs
    ) -> Order:
        """创建订单"""
        order = Order(
            order_id=self._generate_order_id(),
            symbol=symbol,
            side=side,
            order_type=order_type,
            quantity=quantity,
            price=price,
            stop_price=stop_price,
            status=OrderStatus.PENDING,
            created_at=datetime.now(),
            **kwargs
        )

        self.orders[order.order_id] = order
        self.pending_orders.append(order)

        logger.info(f"Order created: {order.order_id} - {side.value} {quantity} {symbol}")
        return order

    def submit_order(self, order_id: str) -> Order:
        """提交订单"""
        order = self.orders.get(order_id)
        if not order:
            raise ValueError(f"Order not found: {order_id}")

        if order.status != OrderStatus.PENDING:
            raise ValueError(f"Order cannot be submitted: {order.status}")

        order.status = OrderStatus.SUBMITTED
        order.submitted_at = datetime.now()

        self.pending_orders.remove(order)
        self.active_orders.append(order)

        logger.info(f"Order submitted: {order.order_id}")
        return order

    def fill_order(
        self,
        order_id: str,
        fill_quantity: int,
        fill_price: float,
        commission: float = 0.0
    ) -> Order:
        """成交订单"""
        order = self.orders.get(order_id)
        if not order:
            raise ValueError(f"Order not found: {order_id}")

        # 更新成交信息
        total_filled = order.filled_quantity + fill_quantity
        order.avg_fill_price = (
            (order.avg_fill_price * order.filled_quantity + fill_price * fill_quantity)
            / total_filled
        )
        order.filled_quantity = total_filled
        order.commission += commission

        # 更新订单状态
        if order.filled_quantity >= order.quantity:
            order.status = OrderStatus.FILLED
            order.filled_at = datetime.now()
            self.active_orders.remove(order)
            self.completed_orders.append(order)
        else:
            order.status = OrderStatus.PARTIAL_FILLED

        logger.info(f"Order filled: {order.order_id} - {fill_quantity} @ {fill_price}")
        return order

    def cancel_order(self, order_id: str, reason: str = "") -> Order:
        """取消订单"""
        order = self.orders.get(order_id)
        if not order:
            raise ValueError(f"Order not found: {order_id}")

        if order.status in [OrderStatus.FILLED, OrderStatus.CANCELLED, OrderStatus.REJECTED]:
            raise ValueError(f"Order cannot be cancelled: {order.status}")

        order.status = OrderStatus.CANCELLED
        order.cancelled_at = datetime.now()
        order.rejected_reason = reason

        if order in self.pending_orders:
            self.pending_orders.remove(order)
        if order in self.active_orders:
            self.active_orders.remove(order)

        self.completed_orders.append(order)

        logger.info(f"Order cancelled: {order.order_id} - {reason}")
        return order

    def get_order(self, order_id: str) -> Optional[Order]:
        """获取订单"""
        return self.orders.get(order_id)

    def get_orders_by_symbol(self, symbol: str) -> List[Order]:
        """获取指定股票的所有订单"""
        return [o for o in self.orders.values() if o.symbol == symbol]

    def get_active_orders(self) -> List[Order]:
        """获取活跃订单"""
        return self.active_orders.copy()

    def _generate_order_id(self) -> str:
        """生成订单ID"""
        import uuid
        timestamp = datetime.now().strftime("%Y%m%d%H%M%S")
        return f"ORD{timestamp}{uuid.uuid4().hex[:6].upper()}"
```

### 3.2 持仓管理

```python
@dataclass
class Position:
    """持仓模型"""
    symbol: str
    quantity: int                   # 持仓数量
    available_quantity: int         # 可用数量
    avg_cost: float                 # 平均成本
    current_price: float            # 当前价格
    market_value: float             # 市值
    cost_basis: float               # 成本
    unrealized_pnl: float          # 未实现盈亏
    realized_pnl: float            # 已实现盈亏
    total_pnl: float               # 总盈亏
    pnl_pct: float                 # 盈亏比例

    # 止损止盈
    stop_loss: Optional[float] = None
    take_profit: Optional[float] = None

    # 时间
    first_buy_at: Optional[datetime] = None
    last_update_at: Optional[datetime] = None

class PositionManager:
    """持仓管理器"""

    def __init__(self):
        self.positions: Dict[str, Position] = {}
        self.closed_positions: List[Position] = []

    def add_position(
        self,
        symbol: str,
        quantity: int,
        price: float,
        timestamp: datetime
    ):
        """增加持仓"""
        if symbol in self.positions:
            pos = self.positions[symbol]
            # 更新平均成本
            total_quantity = pos.quantity + quantity
            pos.avg_cost = (pos.avg_cost * pos.quantity + price * quantity) / total_quantity
            pos.quantity = total_quantity
            pos.available_quantity += quantity
            pos.cost_basis = pos.avg_cost * pos.quantity
        else:
            # 创建新持仓
            self.positions[symbol] = Position(
                symbol=symbol,
                quantity=quantity,
                available_quantity=quantity,
                avg_cost=price,
                current_price=price,
                market_value=price * quantity,
                cost_basis=price * quantity,
                unrealized_pnl=0.0,
                realized_pnl=0.0,
                total_pnl=0.0,
                pnl_pct=0.0,
                first_buy_at=timestamp,
                last_update_at=timestamp
            )

        logger.info(f"Position added: {symbol} +{quantity} @ {price}")

    def reduce_position(
        self,
        symbol: str,
        quantity: int,
        price: float,
        timestamp: datetime
    ) -> float:
        """减少持仓,返回已实现盈亏"""
        if symbol not in self.positions:
            raise ValueError(f"No position for {symbol}")

        pos = self.positions[symbol]

        if quantity > pos.available_quantity:
            raise ValueError(f"Insufficient available quantity: {quantity} > {pos.available_quantity}")

        # 计算已实现盈亏
        realized_pnl = (price - pos.avg_cost) * quantity
        pos.realized_pnl += realized_pnl

        # 更新持仓
        pos.quantity -= quantity
        pos.available_quantity -= quantity
        pos.cost_basis = pos.avg_cost * pos.quantity
        pos.last_update_at = timestamp

        # 如果持仓清空
        if pos.quantity == 0:
            self.closed_positions.append(pos)
            del self.positions[symbol]
            logger.info(f"Position closed: {symbol}")
        else:
            logger.info(f"Position reduced: {symbol} -{quantity} @ {price}")

        return realized_pnl

    def update_market_price(self, symbol: str, price: float):
        """更新市场价格"""
        if symbol in self.positions:
            pos = self.positions[symbol]
            pos.current_price = price
            pos.market_value = price * pos.quantity
            pos.unrealized_pnl = (price - pos.avg_cost) * pos.quantity
            pos.total_pnl = pos.unrealized_pnl + pos.realized_pnl
            pos.pnl_pct = pos.unrealized_pnl / pos.cost_basis if pos.cost_basis > 0 else 0.0
            pos.last_update_at = datetime.now()

    def get_position(self, symbol: str) -> Optional[Position]:
        """获取持仓"""
        return self.positions.get(symbol)

    def get_all_positions(self) -> List[Position]:
        """获取所有持仓"""
        return list(self.positions.values())

    def get_total_market_value(self) -> float:
        """获取总市值"""
        return sum(pos.market_value for pos in self.positions.values())

    def get_total_cost(self) -> float:
        """获取总成本"""
        return sum(pos.cost_basis for pos in self.positions.values())

    def get_total_pnl(self) -> float:
        """获取总盈亏"""
        return sum(pos.total_pnl for pos in self.positions.values())

    def check_stop_loss_triggered(self) -> List[str]:
        """检查止损触发"""
        triggered = []
        for symbol, pos in self.positions.items():
            if pos.stop_loss and pos.current_price <= pos.stop_loss:
                triggered.append(symbol)
        return triggered

    def check_take_profit_triggered(self) -> List[str]:
        """检查止盈触发"""
        triggered = []
        for symbol, pos in self.positions.items():
            if pos.take_profit and pos.current_price >= pos.take_profit:
                triggered.append(symbol)
        return triggered
```

### 3.3 账户管理

```python
@dataclass
class Account:
    """账户模型"""
    account_id: str
    initial_capital: float
    cash: float
    position_value: float
    total_value: float
    available_cash: float          # 可用资金
    frozen_cash: float             # 冻结资金

    # 统计
    total_pnl: float
    daily_pnl: float
    commission_paid: float

    # 风控
    daily_loss: float
    max_daily_loss: float
    consecutive_losses: int
    is_trading_halted: bool = False

class AccountManager:
    """账户管理器"""

    def __init__(self, initial_capital: float, max_daily_loss_pct: float = 0.03):
        self.account = Account(
            account_id=self._generate_account_id(),
            initial_capital=initial_capital,
            cash=initial_capital,
            position_value=0.0,
            total_value=initial_capital,
            available_cash=initial_capital,
            frozen_cash=0.0,
            total_pnl=0.0,
            daily_pnl=0.0,
            commission_paid=0.0,
            daily_loss=0.0,
            max_daily_loss=initial_capital * max_daily_loss_pct,
            consecutive_losses=0
        )

        self.daily_records = []

    def freeze_cash(self, amount: float):
        """冻结资金（下单时）"""
        if amount > self.account.available_cash:
            raise ValueError(f"Insufficient available cash: {amount} > {self.account.available_cash}")

        self.account.available_cash -= amount
        self.account.frozen_cash += amount
        logger.debug(f"Cash frozen: {amount}")

    def unfreeze_cash(self, amount: float):
        """解冻资金（订单取消时）"""
        self.account.frozen_cash -= amount
        self.account.available_cash += amount
        logger.debug(f"Cash unfrozen: {amount}")

    def process_buy(self, amount: float, commission: float):
        """处理买入"""
        total_cost = amount + commission

        if total_cost > self.account.frozen_cash:
            raise ValueError(f"Insufficient frozen cash: {total_cost} > {self.account.frozen_cash}")

        self.account.frozen_cash -= total_cost
        self.account.cash -= total_cost
        self.account.commission_paid += commission

        logger.debug(f"Buy processed: amount={amount}, commission={commission}")

    def process_sell(self, amount: float, commission: float):
        """处理卖出"""
        net_amount = amount - commission

        self.account.cash += net_amount
        self.account.available_cash += net_amount
        self.account.commission_paid += commission

        logger.debug(f"Sell processed: amount={amount}, commission={commission}")

    def update_account_value(self, position_value: float):
        """更新账户价值"""
        self.account.position_value = position_value
        self.account.total_value = self.account.cash + position_value
        self.account.total_pnl = self.account.total_value - self.account.initial_capital

        # 计算日内盈亏
        if self.daily_records:
            prev_value = self.daily_records[-1]["total_value"]
            self.account.daily_pnl = self.account.total_value - prev_value

            # 更新连续亏损计数
            if self.account.daily_pnl < 0:
                self.account.daily_loss += abs(self.account.daily_pnl)
                self.account.consecutive_losses += 1
            else:
                self.account.consecutive_losses = 0

    def reset_daily_stats(self):
        """重置日统计（每日收盘后）"""
        # 记录当日数据
        self.daily_records.append({
            "date": datetime.now().date(),
            "cash": self.account.cash,
            "position_value": self.account.position_value,
            "total_value": self.account.total_value,
            "daily_pnl": self.account.daily_pnl
        })

        # 重置日统计
        self.account.daily_pnl = 0.0
        self.account.daily_loss = 0.0

    def check_trading_halt(self) -> bool:
        """检查是否需要暂停交易"""
        # 条件1：日内亏损超限
        if self.account.daily_loss >= self.account.max_daily_loss:
            self.account.is_trading_halted = True
            logger.warning("Trading halted: daily loss limit reached")
            return True

        # 条件2：连续亏损3次
        if self.account.consecutive_losses >= 3:
            self.account.is_trading_halted = True
            logger.warning("Trading halted: 3 consecutive losses")
            return True

        return False

    def _generate_account_id(self) -> str:
        import uuid
        return f"ACC{uuid.uuid4().hex[:8].upper()}"
```

### 3.4 风险控制

```python
class RiskRule:
    """风控规则基类"""

    def __init__(self, name: str):
        self.name = name

    def check(self, order: Order, account: Account, positions: Dict[str, Position]) -> tuple[bool, str]:
        """检查规则,返回 (是否通过, 拒绝原因)"""
        raise NotImplementedError

class CashSufficientRule(RiskRule):
    """资金充足性检查"""

    def __init__(self):
        super().__init__("cash_sufficient")

    def check(self, order: Order, account: Account, positions: Dict[str, Position]) -> tuple[bool, str]:
        if order.side == OrderSide.BUY:
            estimated_cost = order.quantity * (order.price or 0) * 1.01  # 预留1%佣金
            if estimated_cost > account.available_cash:
                return False, f"Insufficient cash: need {estimated_cost}, have {account.available_cash}"

        return True, ""

class PositionLimitRule(RiskRule):
    """单股仓位限制"""

    def __init__(self, max_position_pct: float = 0.2):
        super().__init__("position_limit")
        self.max_position_pct = max_position_pct

    def check(self, order: Order, account: Account, positions: Dict[str, Position]) -> tuple[bool, str]:
        if order.side == OrderSide.BUY:
            # 计算买入后的仓位占比
            estimated_cost = order.quantity * (order.price or 0)
            current_cost = positions.get(order.symbol).cost_basis if order.symbol in positions else 0
            new_position_pct = (current_cost + estimated_cost) / account.total_value

            if new_position_pct > self.max_position_pct:
                return False, f"Position limit exceeded: {new_position_pct:.1%} > {self.max_position_pct:.1%}"

        return True, ""

class DailyLossLimitRule(RiskRule):
    """日内亏损限制"""

    def __init__(self):
        super().__init__("daily_loss_limit")

    def check(self, order: Order, account: Account, positions: Dict[str, Position]) -> tuple[bool, str]:
        if account.daily_loss >= account.max_daily_loss:
            return False, f"Daily loss limit reached: {account.daily_loss} >= {account.max_daily_loss}"

        return True, ""

class TotalPositionLimitRule(RiskRule):
    """总仓位限制"""

    def __init__(self, max_total_position_pct: float = 0.8):
        super().__init__("total_position_limit")
        self.max_total_position_pct = max_total_position_pct

    def check(self, order: Order, account: Account, positions: Dict[str, Position]) -> tuple[bool, str]:
        if order.side == OrderSide.BUY:
            current_position_pct = account.position_value / account.total_value if account.total_value > 0 else 0
            estimated_cost = order.quantity * (order.price or 0)
            new_position_pct = (account.position_value + estimated_cost) / account.total_value

            if new_position_pct > self.max_total_position_pct:
                return False, f"Total position limit exceeded: {new_position_pct:.1%} > {self.max_total_position_pct:.1%}"

        return True, ""

class StopLossRule(RiskRule):
    """强制止损检查"""

    def __init__(self):
        super().__init__("stop_loss")

    def check(self, order: Order, account: Account, positions: Dict[str, Position]) -> tuple[bool, str]:
        # 这是一个主动检查,不是订单检查
        # 在每次更新价格后调用
        return True, ""

class RiskManager:
    """风险管理器"""

    def __init__(self, config: Dict):
        self.config = config
        self.rules = [
            CashSufficientRule(),
            PositionLimitRule(config.get("max_position_pct", 0.2)),
            TotalPositionLimitRule(config.get("max_total_position_pct", 0.8)),
            DailyLossLimitRule()
        ]

    def check_order(
        self,
        order: Order,
        account: Account,
        positions: Dict[str, Position]
    ) -> tuple[bool, str]:
        """检查订单是否符合风控规则"""

        # 检查交易是否被暂停
        if account.is_trading_halted:
            return False, "Trading is halted"

        # 依次检查所有规则
        for rule in self.rules:
            passed, reason = rule.check(order, account, positions)
            if not passed:
                logger.warning(f"Risk check failed: {rule.name} - {reason}")
                return False, f"[{rule.name}] {reason}"

        order.risk_checked = True
        return True, ""

    def check_stop_loss_triggered(
        self,
        positions: Dict[str, Position]
    ) -> List[str]:
        """检查止损触发"""
        triggered = []

        for symbol, pos in positions.items():
            # 检查自定义止损
            if pos.stop_loss and pos.current_price <= pos.stop_loss:
                triggered.append(symbol)
                logger.warning(f"Stop loss triggered: {symbol} @ {pos.current_price} <= {pos.stop_loss}")

            # 检查默认止损 (-5%)
            elif pos.pnl_pct <= -0.05:
                triggered.append(symbol)
                logger.warning(f"Default stop loss triggered: {symbol} @ {pos.pnl_pct:.2%}")

        return triggered
```

### 3.5 交易执行器

```python
class PaperTrader:
    """模拟交易器"""

    def __init__(self, data_engine):
        self.data_engine = data_engine

    def execute_order(self, order: Order, execution_date: datetime) -> tuple[float, float]:
        """执行订单,返回 (成交价格, 佣金)"""

        # 获取开盘价（模拟次日开盘成交）
        df = self.data_engine.get_daily_data(
            symbol=order.symbol,
            start_date=(execution_date - timedelta(days=5)).strftime("%Y-%m-%d"),
            end_date=execution_date.strftime("%Y-%m-%d")
        )

        if execution_date.date() not in df.index.date:
            raise ValueError(f"No data for {order.symbol} on {execution_date.date()}")

        execution_price = df.loc[df.index.date == execution_date.date()].iloc[0]["open"]

        # 应用滑点
        slippage = 0.001  # 0.1%
        if order.side == OrderSide.BUY:
            execution_price *= (1 + slippage)
        else:
            execution_price *= (1 - slippage)

        # 计算佣金
        amount = execution_price * order.quantity
        commission = max(amount * 0.00025, 5.0)  # 万2.5,最低5元

        # 卖出加印花税
        if order.side == OrderSide.SELL:
            commission += amount * 0.001

        logger.info(f"Paper trade executed: {order.order_id} @ {execution_price}")

        return execution_price, commission

class TradeExecutor:
    """交易执行器（统一接口）"""

    def __init__(self, mode: str, config: Dict):
        self.mode = mode  # "paper" or "live"
        self.config = config

        if mode == "paper":
            self.executor = PaperTrader(config["data_engine"])
        elif mode == "live":
            broker_type = config.get("broker", "futu")
            if broker_type == "futu":
                self.executor = FutuBroker(config)
            elif broker_type == "alpaca":
                self.executor = AlpacaBroker(config)
            else:
                raise ValueError(f"Unknown broker: {broker_type}")

    def execute(self, order: Order, execution_date: datetime = None) -> tuple[float, float]:
        """执行订单"""
        return self.executor.execute_order(order, execution_date or datetime.now())
```

## 4. 交易引擎主类

```python
class TradeEngine:
    """交易引擎主类"""

    def __init__(self, config: Dict):
        self.config = config

        # 初始化组件
        self.order_manager = OrderManager()
        self.position_manager = PositionManager()
        self.account_manager = AccountManager(
            initial_capital=config.get("initial_capital", 1000000),
            max_daily_loss_pct=config.get("max_daily_loss_pct", 0.03)
        )
        self.risk_manager = RiskManager(config.get("risk", {}))
        self.executor = TradeExecutor(
            mode=config.get("mode", "paper"),
            config=config
        )

        # 交易日志
        self.trade_logger = TradeLogger(config.get("log_path", "./logs/trades"))

    def submit_order(
        self,
        symbol: str,
        side: OrderSide,
        quantity: int,
        order_type: OrderType = OrderType.MARKET,
        price: Optional[float] = None,
        **kwargs
    ) -> Order:
        """提交订单"""

        # 1. 创建订单
        order = self.order_manager.create_order(
            symbol=symbol,
            side=side,
            quantity=quantity,
            order_type=order_type,
            price=price,
            **kwargs
        )

        # 2. 风控检查
        passed, reason = self.risk_manager.check_order(
            order=order,
            account=self.account_manager.account,
            positions=self.position_manager.positions
        )

        if not passed:
            order.status = OrderStatus.REJECTED
            order.rejected_reason = reason
            self.trade_logger.log_order_rejected(order, reason)
            logger.warning(f"Order rejected: {order.order_id} - {reason}")
            return order

        # 3. 冻结资金（买入时）
        if side == OrderSide.BUY:
            estimated_cost = quantity * (price or 0) * 1.01
            self.account_manager.freeze_cash(estimated_cost)

        # 4. 提交订单
        self.order_manager.submit_order(order.order_id)
        self.trade_logger.log_order_submitted(order)

        return order

    def execute_pending_orders(self, execution_date: datetime = None):
        """执行待成交订单"""
        execution_date = execution_date or datetime.now()

        for order in self.order_manager.active_orders[:]:
            try:
                # 执行订单
                execution_price, commission = self.executor.execute(order, execution_date)

                # 更新订单状态
                self.order_manager.fill_order(
                    order_id=order.order_id,
                    fill_quantity=order.quantity,
                    fill_price=execution_price,
                    commission=commission
                )

                # 更新账户和持仓
                amount = execution_price * order.quantity

                if order.side == OrderSide.BUY:
                    self.account_manager.process_buy(amount, commission)
                    self.position_manager.add_position(
                        symbol=order.symbol,
                        quantity=order.quantity,
                        price=execution_price,
                        timestamp=execution_date
                    )
                else:
                    self.account_manager.process_sell(amount, commission)
                    realized_pnl = self.position_manager.reduce_position(
                        symbol=order.symbol,
                        quantity=order.quantity,
                        price=execution_price,
                        timestamp=execution_date
                    )

                # 记录日志
                self.trade_logger.log_trade_executed(order, execution_price, commission)

            except Exception as e:
                logger.error(f"Failed to execute order {order.order_id}: {e}")
                self.order_manager.cancel_order(order.order_id, reason=str(e))

    def update_market_prices(self, prices: Dict[str, float]):
        """更新市场价格"""
        for symbol, price in prices.items():
            self.position_manager.update_market_price(symbol, price)

        # 更新账户价值
        position_value = self.position_manager.get_total_market_value()
        self.account_manager.update_account_value(position_value)

        # 检查止损触发
        triggered = self.risk_manager.check_stop_loss_triggered(self.position_manager.positions)
        for symbol in triggered:
            pos = self.position_manager.get_position(symbol)
            if pos:
                # 自动创建平仓订单
                self.submit_order(
                    symbol=symbol,
                    side=OrderSide.SELL,
                    quantity=pos.quantity,
                    reason="Stop loss triggered"
                )

    def get_account_status(self) -> Dict:
        """获取账户状态"""
        return {
            "cash": self.account_manager.account.cash,
            "available_cash": self.account_manager.account.available_cash,
            "position_value": self.account_manager.account.position_value,
            "total_value": self.account_manager.account.total_value,
            "total_pnl": self.account_manager.account.total_pnl,
            "daily_pnl": self.account_manager.account.daily_pnl,
            "is_trading_halted": self.account_manager.account.is_trading_halted
        }

    def get_positions(self) -> List[Dict]:
        """获取持仓列表"""
        return [
            {
                "symbol": pos.symbol,
                "quantity": pos.quantity,
                "avg_cost": pos.avg_cost,
                "current_price": pos.current_price,
                "market_value": pos.market_value,
                "unrealized_pnl": pos.unrealized_pnl,
                "pnl_pct": pos.pnl_pct
            }
            for pos in self.position_manager.get_all_positions()
        ]

    def get_orders(self, status: Optional[OrderStatus] = None) -> List[Order]:
        """获取订单列表"""
        if status:
            return [o for o in self.order_manager.orders.values() if o.status == status]
        return list(self.order_manager.orders.values())
```

## 7. 总结

交易引擎设计要点:
1. **风控第一**: 多层风控保护,硬编码不可绕过
2. **状态精确**: 订单、持仓状态精确跟踪
3. **模拟真实**: 模拟交易尽可能还原真实环境
4. **易于切换**: 模拟与实盘无缝切换
5. **完整日志**: 所有交易操作可审计

交易引擎是资金安全的最后一道防线,必须谨慎设计、充分测试。

# 前端架构设计文档

## 1. 模块概述

### 1.1 核心职责
前端负责提供交互式的量化交易界面:
- 数据可视化(K线图、指标、资金曲线)
- 信号展示与筛选
- 回测配置与结果展示
- 交易操作界面
- 实时数据更新

### 1.2 设计目标
- **专业体验**: 符合金融交易软件标准
- **实时响应**: WebSocket 实时数据推送
- **交互流畅**: 图表缩放、拖拽、交叉光标
- **响应式设计**: 支持桌面
- **赛博主题**: 充满科幻主题的专业配色

## 2. 技术栈

### 2.1 核心技术
- **框架**: React 18 (Hooks + Functional Components)
- **状态管理**: Zustand (轻量级)
- **样式**: TailwindCSS + CSS Modules
- **图表**: Lightweight Charts (TradingView开源)
- **HTTP**: Axios
- **WebSocket**: native WebSocket API
- **路由**: React Router v6
- **构建**: Vite

### 2.2 开发工具
- **TypeScript**: 类型安全
- **ESLint + Prettier**: 代码规范
- **Vitest**: 单元测试
- **Storybook**: 组件文档

## 4. 核心组件设计

### 4.1 K线图组件

```typescript
// components/charts/CandlestickChart.tsx
import React, { useEffect, useRef } from 'react';
import { createChart, IChartApi, ISeriesApi, CandlestickData } from 'lightweight-charts';

interface CandlestickChartProps {
  data: CandlestickData[];
  indicators?: IndicatorData[];
  signals?: SignalData[];
  width?: number;
  height?: number;
  onCrosshairMove?: (price: number, time: number) => void;
}

export const CandlestickChart: React.FC<CandlestickChartProps> = ({
  data,
  indicators,
  signals,
  width = 800,
  height = 600,
  onCrosshairMove
}) => {
  const chartContainerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<IChartApi | null>(null);
  const candlestickSeriesRef = useRef<ISeriesApi<"Candlestick"> | null>(null);

  useEffect(() => {
    if (!chartContainerRef.current) return;

    // 创建图表
    const chart = createChart(chartContainerRef.current, {
      width,
      height,
      layout: {
        background: { color: '#1e1e1e' },
        textColor: '#d1d5db',
      },
      grid: {
        vertLines: { color: '#2b2b2b' },
        horzLines: { color: '#2b2b2b' },
      },
      crosshair: {
        mode: 1, // CrosshairMode.Normal
      },
      rightPriceScale: {
        borderColor: '#2b2b2b',
      },
      timeScale: {
        borderColor: '#2b2b2b',
        timeVisible: true,
        secondsVisible: false,
      },
    });

    // 添加K线系列
    const candlestickSeries = chart.addCandlestickSeries({
      upColor: '#26a69a',
      downColor: '#ef5350',
      borderVisible: false,
      wickUpColor: '#26a69a',
      wickDownColor: '#ef5350',
    });

    candlestickSeries.setData(data);

    chartRef.current = chart;
    candlestickSeriesRef.current = candlestickSeries;

    // 十字光标移动事件
    chart.subscribeCrosshairMove((param) => {
      if (param.time && param.seriesPrices.size > 0) {
        const price = param.seriesPrices.get(candlestickSeries);
        if (price && onCrosshairMove) {
          onCrosshairMove(price as number, param.time as number);
        }
      }
    });

    // 自适应大小
    const handleResize = () => {
      chart.applyOptions({ width: chartContainerRef.current?.clientWidth });
    };
    window.addEventListener('resize', handleResize);

    return () => {
      window.removeEventListener('resize', handleResize);
      chart.remove();
    };
  }, []);

  // 更新数据
  useEffect(() => {
    if (candlestickSeriesRef.current) {
      candlestickSeriesRef.current.setData(data);
    }
  }, [data]);

  // 添加指标
  useEffect(() => {
    if (!chartRef.current || !indicators) return;

    indicators.forEach((indicator) => {
      const lineSeries = chartRef.current!.addLineSeries({
        color: indicator.color,
        lineWidth: 2,
        title: indicator.name,
      });
      lineSeries.setData(indicator.data);
    });
  }, [indicators]);

  // 标注信号
  useEffect(() => {
    if (!candlestickSeriesRef.current || !signals) return;

    const markers = signals.map((signal) => ({
      time: signal.time,
      position: signal.type === 'BUY' ? 'belowBar' : 'aboveBar',
      color: signal.type === 'BUY' ? '#26a69a' : '#ef5350',
      shape: signal.type === 'BUY' ? 'arrowUp' : 'arrowDown',
      text: `${signal.type} @ ${signal.price.toFixed(2)}`,
    }));

    candlestickSeriesRef.current.setMarkers(markers);
  }, [signals]);

  return (
    <div className="chart-container">
      <div ref={chartContainerRef} />
    </div>
  );
};
```

```typescript
import { create } from 'zustand';
import { MarketData, Kline } from '@/types/market';

interface MarketStore {
  // 状态
  currentSymbol: string;
  klineData: Kline[];
  indicators: any[];

  // 操作
  setCurrentSymbol: (symbol: string) => void;

  addIndicator: (indicator: any) => void;
  removeIndicator: (name: string) => void;
  setLoading: (loading: boolean) => void;

}

export const useMarketStore = create<MarketStore>((set) => ({
  currentSymbol: '000001.SZ',
  klineData: [],
  indicators: [],
  loading: false,
  error: null,

  setCurrentSymbol: (symbol) => set({ currentSymbol: symbol }),
  setKlineData: (data) => set({ klineData: data }),
  addIndicator: (indicator) =>
    set((state) => ({ indicators: [...state.indicators, indicator] })),
  removeIndicator: (name) =>
    set((state) => ({
      indicators: state.indicators.filter((ind) => ind.name !== name),
    })),
  setLoading: (loading) => set({ loading }),
  setError: (error) => set({ error }),
}));
```

```typescript
// stores/signalStore.ts
import { create } from 'zustand';
import { Signal } from '@/types/signal';

interface SignalStore {
  signals: Signal[];
  filters: {
    minStrength: number;
    signalTypes: string[];
    dateRange: [string, string] | null;
  };

  setSignals: (signals: Signal[]) => void;
  addSignal: (signal: Signal) => void;
  updateFilters: (filters: Partial<SignalStore['filters']>) => void;
  getFilteredSignals: () => Signal[];
}

export const useSignalStore = create<SignalStore>((set, get) => ({
  signals: [],
  filters: {
    minStrength: 0.5,
    signalTypes: ['BUY', 'SELL'],
    dateRange: null,
  },

  setSignals: (signals) => set({ signals }),
  addSignal: (signal) =>
    set((state) => ({ signals: [...state.signals, signal] })),
  updateFilters: (filters) =>
    set((state) => ({ filters: { ...state.filters, ...filters } })),

  getFilteredSignals: () => {
    const { signals, filters } = get();
    return signals.filter((signal) => {
      if (signal.strength < filters.minStrength) return false;
      if (!filters.signalTypes.includes(signal.signal_type)) return false;
      // 日期范围过滤...
      return true;
    });
  },
}));
```

### 6.1 WebSocket Hook

```typescript
// hooks/useWebSocket.ts
import { useEffect, useRef, useState } from 'react';

interface UseWebSocketOptions {
  onMessage?: (data: any) => void;
  onError?: (error: Event) => void;
  onOpen?: () => void;
  onClose?: () => void;
  reconnect?: boolean;
  reconnectInterval?: number;
}

export const useWebSocket = (url: string, options: UseWebSocketOptions = {}) => {
  const [isConnected, setIsConnected] = useState(false);
  const [lastMessage, setLastMessage] = useState<any>(null);
  const wsRef = useRef<WebSocket | null>(null);
  const reconnectTimeoutRef = useRef<NodeJS.Timeout>();

  const connect = () => {
    try {
      const ws = new WebSocket(url);

      ws.onopen = () => {
        console.log('WebSocket connected');
        setIsConnected(true);
        options.onOpen?.();
      };

      ws.onmessage = (event) => {
        const data = JSON.parse(event.data);
        setLastMessage(data);
        options.onMessage?.(data);
      };

      ws.onerror = (error) => {
        console.error('WebSocket error:', error);
        options.onError?.(error);
      };

      ws.onclose = () => {
        console.log('WebSocket disconnected');
        setIsConnected(false);
        options.onClose?.();

        // 自动重连
        if (options.reconnect !== false) {
          reconnectTimeoutRef.current = setTimeout(() => {
            connect();
          }, options.reconnectInterval || 5000);
        }
      };

      wsRef.current = ws;
    } catch (error) {
      console.error('Failed to connect WebSocket:', error);
    }
  };

  const disconnect = () => {
    if (reconnectTimeoutRef.current) {
      clearTimeout(reconnectTimeoutRef.current);
    }
    wsRef.current?.close();
  };

  const send = (data: any) => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(data));
    }
  };

  useEffect(() => {
    connect();
    return () => disconnect();
  }, [url]);

  return { isConnected, lastMessage, send, disconnect };
};
```

```typescript
// hooks/useMarketData.ts
import { useEffect } from 'react';
import { useMarketStore } from '@/stores/marketStore';
import { marketService } from '@/services/marketService';

export const useMarketData = (symbol: string, startDate: string, endDate: string) => {
  const { setKlineData, setLoading, setError } = useMarketStore();

  useEffect(() => {
    const fetchData = async () => {
      setLoading(true);
      setError(null);

      try {
        const data = await marketService.getKlineData(symbol, startDate, endDate);
        setKlineData(data);
      } catch (error) {
        setError(error instanceof Error ? error.message : 'Failed to fetch data');
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, [symbol, startDate, endDate]);

  return useMarketStore();
};
```

```typescript
  }
);

export default api;

// services/marketService.ts
import api from './api';
import { Kline, RealtimeQuote } from '@/types/market';

export const marketService = {
  getKlineData: async (
    symbol: string,
    startDate: string,
    endDate: string
  ): Promise<Kline[]> => {
    return api.get('/market/kline', {
      params: { symbol, start_date: startDate, end_date: endDate },
    });
  },

  getRealtimeQuote: async (symbols: string[]): Promise<RealtimeQuote[]> => {
    return api.post('/market/realtime', { symbols });
  },

  searchSymbols: async (keyword: string): Promise<any[]> => {
    return api.get('/market/search', { params: { keyword } });
  },
};
```

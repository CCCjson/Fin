# 部署运维设计文档

## 1. 概述

### 1.1 部署目标
- **简单快速**: 一键部署,快速上手
- **稳定可靠**: 7x24小时稳定运行
- **易于维护**: 日志监控,故障排查
- **可扩展**: 支持从本地到云端的迁移

### 1.2 部署模式
- **本地开发**: 笔记本/台式机,开发调试
- **本地生产**: 个人服务器/NAS,长期运行
- **云端部署**: 云服务器,高可用部署

## 2. 本地部署

### 2.1 系统要求

#### 硬件要求
- **CPU**: 4核以上推荐
- **内存**: 8GB以上推荐
- **硬盘**: 50GB以上可用空间(存储历史数据)
- **网络**: 稳定的互联网连接

#### 软件要求
- **操作系统**: Windows 10+, macOS 10.15+, Linux (Ubuntu 20.04+)
- **Python**: 3.12+
- **Node.js**: 18+ (前端开发)
- **Git**: 版本控制

### 2.2 快速部署脚本

#### Linux/macOS 一键部署

```bash
#!/bin/bash
# deploy.sh - 一键部署脚本

set -e  # 遇到错误立即退出

echo "========================================="
echo "  量化交易系统 - 一键部署"
echo "========================================="

# 1. 检查 Python 版本
echo "检查 Python 版本..."
python_version=$(python3 --version | awk '{print $2}')
required_version="3.12"

if [ "$(printf '%s\n' "$required_version" "$python_version" | sort -V | head -n1)" != "$required_version" ]; then
    echo "错误: 需要 Python >= 3.12, 当前版本: $python_version"
    exit 1
fi

# 2. 创建虚拟环境
echo "创建虚拟环境..."
python3 -m venv venv
source venv/bin/activate

# 3. 安装后端依赖
echo "安装后端依赖..."
cd backend
pip install -r requirements.txt

# 4. 初始化数据库
echo "初始化数据库..."
python -c "from data_engine.storage.database import init_db; init_db()"

# 5. 创建配置文件
echo "创建配置文件..."
if [ ! -f ".env" ]; then
    cp .env.example .env
    echo "请编辑 backend/.env 文件,配置 API Key 等信息"
fi

# 6. 安装前端依赖
echo "安装前端依赖..."
cd ../frontend
npm install

# 7. 构建前端
echo "构建前端..."
npm run build

echo "========================================="
echo "  部署完成!"
echo "========================================="
echo ""
echo "启动后端: cd backend && python main.py"
echo "启动前端: cd frontend && npm run dev"
echo ""
```

## 4. 日志管理

### 4.1 日志配置

```python
# backend/logging_config.py
from loguru import logger
import sys
from pathlib import Path

def setup_logging(log_dir: str = "./logs"):
    """配置日志"""

    # 创建日志目录
    Path(log_dir).mkdir(parents=True, exist_ok=True)

    # 移除默认处理器
    logger.remove()

    # 控制台输出 (彩色)
    logger.add(
        sys.stdout,
        format="<green>{time:YYYY-MM-DD HH:mm:ss}</green> | <level>{level: <8}</level> | <cyan>{name}</cyan>:<cyan>{function}</cyan> - <level>{message}</level>",
        level="INFO",
        colorize=True
    )

    # 文件输出 - INFO 级别
    logger.add(
        f"{log_dir}/info.log",
        format="{time:YYYY-MM-DD HH:mm:ss} | {level: <8} | {name}:{function}:{line} - {message}",
        level="INFO",
        rotation="100 MB",  # 日志轮转
        retention="30 days",  # 保留30天
        compression="zip"  # 压缩旧日志
    )

    # 文件输出 - ERROR 级别
    logger.add(
        f"{log_dir}/error.log",
        format="{time:YYYY-MM-DD HH:mm:ss} | {level: <8} | {name}:{function}:{line} - {message}",
        level="ERROR",
        rotation="50 MB",
        retention="90 days",
        compression="zip",
        backtrace=True,  # 包含回溯信息
        diagnose=True  # 包含诊断信息
    )

    # 文件输出 - 交易日志
    logger.add(
        f"{log_dir}/trades.log",
        format="{time:YYYY-MM-DD HH:mm:ss} | {message}",
        level="INFO",
        filter=lambda record: record["extra"].get("type") == "trade",
        rotation="10 MB",
        retention="365 days",
        compression="zip"
    )

    logger.info("Logging configured successfully")
```

### 4.2 日志查看脚本

```bash
#!/bin/bash
# view_logs.sh - 日志查看脚本

LOGS_DIR="./logs"

case "$1" in
    info)
        tail -f "$LOGS_DIR/info.log"
        ;;
    error)
        tail -f "$LOGS_DIR/error.log"
        ;;
    trade)
        tail -f "$LOGS_DIR/trades.log"
        ;;
    all)
        tail -f "$LOGS_DIR"/*.log
        ;;
    search)
        if [ -z "$2" ]; then
            echo "Usage: $0 search <keyword>"
            exit 1
        fi
        grep -r "$2" "$LOGS_DIR"/*.log
        ;;
    *)
        echo "Usage: $0 {info|error|trade|all|search <keyword>}"
        exit 1
        ;;
esac
```

# backup.sh - 自动备份脚本

BACKUP_DIR="./backups"

tar -czf "$BACKUP_DIR/configs_${DATE}.tar.gz" "$CONFIGS_DIR"

# 5. 清理旧备份 (保留7天)
echo "清理旧备份..."
find "$BACKUP_DIR" -name "*.gz" -mtime +7 -delete

echo "备份完成: $DATE"

## 6. 监控告警

### 6.1 健康检查脚本

```python
# scripts/health_check.py
import requests
import smtplib
from email.mime.text import MIMEText
import time

def check_backend_health():
    """检查后端健康状态"""
    try:
        response = requests.get("http://localhost:8000/api/v1/health", timeout=5)
        return response.status_code == 200
    except Exception as e:
        print(f"Backend health check failed: {e}")
        return False

def check_database():
    """检查数据库连接"""
    try:
        import sqlite3
        conn = sqlite3.connect("./data/market.db")
        cursor = conn.cursor()
        cursor.execute("SELECT COUNT(*) FROM stock_info")
        conn.close()
        return True
    except Exception as e:
        print(f"Database check failed: {e}")
        return False

def send_alert(subject: str, message: str):
    """发送告警邮件"""
    # 配置邮件服务器
    smtp_server = "smtp.gmail.com"
    smtp_port = 587
    sender = "your_email@gmail.com"
    password = "your_password"
    receiver = "alert_email@gmail.com"

    msg = MIMEText(message)
    msg["Subject"] = subject
    msg["From"] = sender
    msg["To"] = receiver

    try:
        with smtplib.SMTP(smtp_server, smtp_port) as server:
            server.starttls()
            server.login(sender, password)
            server.send_message(msg)
        print(f"Alert sent: {subject}")
    except Exception as e:
        print(f"Failed to send alert: {e}")

def main():
    """主函数"""
    print("Running health check...")

    # 检查后端
    if not check_backend_health():
        send_alert(
            "【告警】后端服务异常",
            "量化交易系统后端服务无法访问,请立即检查!"
        )

    # 检查数据库
    if not check_database():
        send_alert(
            "【告警】数据库连接异常",
            "量化交易系统数据库连接失败,请立即检查!"
        )

    print("Health check completed")

if __name__ == "__main__":
    main()
```

## 7. 系统监控

### 7.1 资源监控脚本

```python
# scripts/monitor.py
import psutil
import time
from loguru import logger

def monitor_system():
    """监控系统资源"""
    # CPU使用率
    cpu_percent = psutil.cpu_percent(interval=1)

    # 内存使用率
    memory = psutil.virtual_memory()
    memory_percent = memory.percent

    # 磁盘使用率
    disk = psutil.disk_usage('/')
    disk_percent = disk.percent

    # 网络流量
    net_io = psutil.net_io_counters()

    logger.info(
        f"System Monitor - "
        f"CPU: {cpu_percent}% | "
        f"Memory: {memory_percent}% | "
        f"Disk: {disk_percent}% | "
        f"Network: ↓{net_io.bytes_recv/1024/1024:.2f}MB ↑{net_io.bytes_sent/1024/1024:.2f}MB"
    )

    # 告警阈值
    if cpu_percent > 80:
        logger.warning(f"High CPU usage: {cpu_percent}%")

    if memory_percent > 85:
        logger.warning(f"High memory usage: {memory_percent}%")

    if disk_percent > 90:
        logger.warning(f"High disk usage: {disk_percent}%")

if __name__ == "__main__":
    while True:
        monitor_system()
        time.sleep(60)  # 每分钟监控一次
```
